local function isClothingDecal(d)
    if not d or not d:IsA("Decal") then return false end
    local par = d.Parent; if not par then return false end
    if par:IsA("Shirt") or par:IsA("Pants") or par:IsA("TShirt") or par:IsA("Clothing") then return true end
    local pn = string.lower(par.Name or "")
    if pn=="shirt" or pn=="pants" or pn=="tshirt" or pn=="clothing"
        or pn=="classic shirt" or pn=="classic pants" or pn=="classic tshirt" then return true end
    local dn = string.lower(d.Name or "")
    if dn=="shirt" or dn=="pants" or dn=="tshirt" then return true end
    return false
end
local function isAccessoryHandle(part)
    if not part or not part:IsA("BasePart") then return false end
    local par = part.Parent
    if not par then return false end
    return par:IsA("Accessory") or par:IsA("Hat")
end
local function isInsideAccessory(obj)
    if not obj then return false end
    local par = obj.Parent
    while par and par ~= workspace do
        if par:IsA("Accessory") or par:IsA("Hat") then return true end
        par = par.Parent
    end
    return false
end
local function isInsideTool(obj) return obj:FindFirstAncestorWhichIsA("Tool") ~= nil end

local walkMode, savedWalkSpeed = false, M.W.NORMAL
local function applyWalkMode()
    local char = p.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if hum.WalkSpeed == 0 and walkMode then return end
    hum.WalkSpeed = walkMode and M.W.CALM or savedWalkSpeed
end
local function setWalkMode(state, silent)
    walkMode = state; applyWalkMode()
    if not silent then
        pcall(function() StarterGui:SetCore("SendNotification",
            {Title=walkMode and "Походка: СПОКОЙНАЯ" or "Походка: ОБЫЧНАЯ", Text="[H]", Duration=3}) end)
    end
end
local function toggleWalkMode() setWalkMode(not walkMode) end

local function hideForceFields(char)
    if not config.HideForceField then return end
    local function hide(ff) if ff and ff:IsA("ForceField") then pcall(function() ff.Visible=false end) end end
    for _, d in ipairs(char:GetDescendants()) do hide(d) end
    track(char.DescendantAdded:Connect(hide))
end

local function cameraShake(intensity, duration)
    local char = p.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    task.spawn(function()
        local t0 = os.clock()
        while os.clock()-t0 < duration do
            local d = 1 - (os.clock()-t0)/duration
            hum.CameraOffset = Vector3.new(
                (math.random()-0.5)*2*intensity*d,
                (math.random()-0.5)*2*intensity*d,
                (math.random()-0.5)*2*intensity*d*0.5)
            RunService.RenderStepped:Wait()
        end
        hum.CameraOffset = Vector3.new(0,0,0)
    end)
end

local function loadAssetModel(assetId)
    if not assetId then return nil end
    local clean = tostring(assetId):gsub("rbxassetid://", "")
    local ids = {clean, "rbxassetid://" .. clean}
    for _, id in ipairs(ids) do
        local ok, objs = pcall(function() return game:GetObjects(id) end)
        if ok and objs and objs[1] then
            local obj = objs[1]
            if obj:IsA("Tool") then
                local m = Instance.new("Model"); obj.Parent = m; return m
            elseif obj:IsA("Model") then
                return obj
            elseif obj:IsA("BasePart") then
                local m = Instance.new("Model"); obj.Parent = m; return m
            end
        end
    end
    if InsertService then
        local num = tonumber(clean:match("%d+"))
        if num then
            local ok, model = pcall(function() return InsertService:LoadAsset(num) end)
            if ok and model then
                for _, child in ipairs(model:GetChildren()) do
                    if child:IsA("Model") then return child end
                    if child:IsA("Tool") then
                        local m = Instance.new("Model"); child.Parent = m; return m
                    end
                    if child:IsA("BasePart") then
                        local m = Instance.new("Model"); child.Parent = m; return m
                    end
                end
                return model
            end
        end
    end
    warn("[MONO] Не удалось загрузить модель: " .. tostring(assetId))
    return nil
end

local function recolorGunModel(folder)
    if not folder then return end
    for _, part in ipairs(folder:GetDescendants()) do
        if part:IsA("BasePart") then
            local c = part.Color
            if c.R>0.5 and c.G>0.4 and c.B<0.45 and (c.R+c.G)>c.B*2 then
                part.Color = Color3.fromRGB(60,62,75)
                if part.Material==Enum.Material.Neon then part.Material=Enum.Material.Metal end
            end
        end
    end
end

local function buildGunInstance(def)
    local gc = def.config
    local gun = {def=def, config=gc, tool=nil, handle=nil, muzzle=nil, bodyFolder=nil, welds={}, muzzleWeld=nil}
    local t = Instance.new("Tool")
    t.Name=def.name; t.RequiresHandle=true; t.CanBeDropped=false
    local h = Instance.new("Part")
    h.Name="Handle"; h.Size=Vector3.new(0.2,0.2,0.2); h.Transparency=1
    h.CanCollide=false; h.CanTouch=false; h.CanQuery=false; h.Massless=true
    h.Parent=t
    local snd = Instance.new("Sound"); snd.Name="ShotSound"; snd.SoundId=M.SND.SPAWN; snd.Volume=1.2; snd.Parent=h
    t.PrimaryPart = h
    gun.tool=t; gun.handle=h
    local folder = Instance.new("Folder"); folder.Name="GunBody"; folder.Parent=t
    gun.bodyFolder = folder
    local src = loadAssetModel(def.assetId)
    if src then
        if not src:IsA("Model") then
            local w = Instance.new("Model"); src.Parent = w; src = w
        end
        local sp = src:GetPivot()
        for _, o in ipairs(src:GetDescendants()) do
            if o:IsA("Script") or o:IsA("LocalScript") or o:IsA("ModuleScript") then o:Destroy() end
        end
        for _, part in ipairs(src:GetDescendants()) do
            if part:IsA("BasePart") then
                local rel = sp:ToObjectSpace(part.CFrame)
                local bs = part.Size
                part.Anchored=false; part.CanCollide=false
                part.CanTouch=false; part.CanQuery=false; part.Massless=true
                part.Parent = folder
                local weld = Instance.new("Weld")
                weld.Part0=h; weld.Part1=part
                weld.C0 = gunCFrame(gc)*rel; weld.C1 = CFrame.new()
                weld.Parent=h
                table.insert(gun.welds, {weld=weld, baseCF=rel, part=part, baseSize=bs})
            end
        end
        src:Destroy()
    end
    if def.recolor then recolorGunModel(folder) end
    if #gun.welds == 0 then
        local function mkFb(name, size, color, material, relCF)
            local fb = Instance.new("Part")
            fb.Name=name; fb.Size=size
            fb.Color=color; fb.Material=material or Enum.Material.Metal
            fb.Anchored=false; fb.CanCollide=false
            fb.CanTouch=false; fb.CanQuery=false; fb.Massless=true
            fb.Parent=folder
            local weld = Instance.new("Weld")
            weld.Part0=h; weld.Part1=fb
            weld.C0 = gunCFrame(gc)*relCF; weld.C1=CFrame.new()
            weld.Parent=h
            table.insert(gun.welds, {weld=weld, baseCF=relCF, part=fb, baseSize=size})
        end
        mkFb("FbBody", Vector3.new(0.55,0.55,1.5), Color3.fromRGB(60,62,75), Enum.Material.Metal, CFrame.new(0,0,-0.4))
        mkFb("FbBarrel", Vector3.new(0.32,0.32,1.6), Color3.fromRGB(75,78,92), Enum.Material.Metal, CFrame.new(0,0,-1.7))

local function applyGunConfig(gun)
    gun = gun or getActiveGun()
    if not gun or not gun.welds then return end
    local gc = gun.config
    local s = gc.scale
    local gcf = gunCFrame(gc)
    for _, e in ipairs(gun.welds) do
        if e.weld and e.weld.Parent then
            local sc = CFrame.new(e.baseCF.Position*s)*(e.baseCF-e.baseCF.Position)
            e.weld.C0 = gcf*sc
            if e.part and e.part.Parent and e.baseSize then e.part.Size = e.baseSize*s end
        end
    end
    if gun.muzzleWeld and gun.muzzleWeld.Parent then
        local mo = gun.def.muzzleOffset or CFrame.new(0,0,-1)
        local sm = CFrame.new(mo.Position*s)*(mo-mo.Position)
        gun.muzzleWeld.C0 = gcf*sm
    end
end

    return CFrame.lookAt(from, to, up)
end

local function catmull(p0,p1,p2,p3,u)
    local u2,u3 = u*u, u*u*u
    return 0.5*((2*p1)+(-p0+p2)*u+(2*p0-5*p1+4*p2-p3)*u2+(-p0+3*p1-3*p2+p3)*u3)
end
local function sampleKeys(keys, t)
    local n = #keys
    if t <= keys[1][1] then return keys[1][2],keys[1][3],keys[1][4] end
    if t >= keys[n][1] then return keys[n][2],keys[n][3],keys[n][4] end
    for i=1,n-1 do
        local k1,k2 = keys[i],keys[i+1]
        if t>=k1[1] and t<=k2[1] then
            local k0=keys[math.max(i-1,1)]
            local k3=keys[math.min(i+2,n)]
            local u=(t-k1[1])/(k2[1]-k1[1])
            return catmull(k0[2],k1[2],k2[2],k3[2],u),
                   catmull(k0[3],k1[3],k2[3],k3[3],u),
                   catmull(k0[4],k1[4],k2[4],k3[4],u)
        end
    end
    return 0,0,0
end
local function applyJointRot(joint, cache, rx, ry, rz)
    if not joint or not joint.Parent then return end
    local c = cache[joint]
    if not c then
        local c0 = joint.C0
        local rc = c0 - c0.Position
        c = {rc=rc, rcInv=rc:Inverse()}; cache[joint] = c
    end
    local R = CFrame.Angles(math.rad(rx), math.rad(ry), math.rad(rz))
    joint.Transform = c.rcInv * R * c.rc
end
local function resetJoint(j) if j and j.Parent then j.Transform = CFrame.new() end end

local function playEmote()
    if emoteActive or flight.active then return end
    local char = p.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    local joints = collectEmoteJoints(char)
    if not joints.shoulder then return end
    emoteActive = true; emoteToken = emoteToken + 1
    local token = emoteToken
    local cache = {}
    local total = M.EK.SHOULDER[#M.EK.SHOULDER][1]
    local t0 = os.clock()
    local conns = {}
    local function finish()
        for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
        for _, j in pairs(joints) do resetJoint(j) end
        if token == emoteToken then emoteActive = false; lastMoveStateAnim = nil end
    end
    local function step()
        local t = (os.clock()-t0)*M.EK.SPEED
        if token ~= emoteToken or not joints.shoulder.Parent or hum.Health<=0 or t>=total then
            finish(); return
        end
        local sx,sy,sz = sampleKeys(M.EK.SHOULDER, t)
        applyJointRot(joints.shoulder, cache, sx,sy,sz)
        if joints.elbow then
            local ex,ey,ez = sampleKeys(M.EK.ELBOW, t)
            applyJointRot(joints.elbow, cache, ex,ey,ez)
        end
        if joints.wrist then
            local wx,wy,wz = sampleKeys(M.EK.WRIST, t)
            applyJointRot(joints.wrist, cache, wx,wy,wz)
        end
        if M.EK.HEAD_TRACK and joints.neck then
            local nx,ny,nz = sampleKeys(M.EK.NECK, t)
            applyJointRot(joints.neck, cache, nx,ny,nz)
        end
        if M.EK.TORSO_LEAN and joints.torso then
            local tx,ty,tz = sampleKeys(M.EK.TORSO, t)
            applyJointRot(joints.torso, cache, tx,ty,tz)
        end
    end
    table.insert(conns, RunService.Stepped:Connect(step))
    table.insert(conns, RunService.Heartbeat:Connect(step))
    table.insert(conns, RunService.RenderStepped:Connect(step))
    for _, c in ipairs(conns) do track(c) end
end

local function gunRaiseAmount()
    local dt = os.clock() - gunShotT
    if dt < 0 then return 0 end
    local UP, HOLD, DOWN = 0.12, 0.30, 0.35
    if dt < UP then return smooth01(dt/UP)
    elseif dt < UP+HOLD then return 1
    elseif dt < UP+HOLD+DOWN then return 1 - smooth01((dt-UP-HOLD)/DOWN)
    end
    return 0
end

local function buildFlyModel(hrp)
    local src = getFlyModelSrc()
    if not src then return nil end
    local cfg = getgenv().FlyModelConfig
    local model = src:Clone()
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript")
            or d:IsA("JointInstance") or d:IsA("WeldConstraint") then
            d:Destroy()
        end
    end
    local parts = {}
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("BasePart") then table.insert(parts, d) end
    end
    if #parts == 0 then model:Destroy(); return nil end
    model.Name = "MonoFlyModel"
    model.Parent = workspace
    for _, pt in ipairs(parts) do
        pt.Anchored = true; pt.CanCollide = false; pt.CanTouch = false; pt.CanQuery = false
        pt.Massless = true
    end
    pcall(function() model:ScaleTo(cfg.scale or 1) end)

    local primary, bestVol = parts[1], -1
    for _, pt in ipairs(parts) do
        local v = pt.Size.X * pt.Size.Y * pt.Size.Z
        if v > bestVol then bestVol = v; primary = pt end
    end
    pcall(function() model.PrimaryPart = primary end)

    local bbCF, bbSize = model:GetBoundingBox()
    local pivotToBB = model:GetPivot():ToObjectSpace(bbCF)

    local function computePivotInHrp()
        local c = getgenv().FlyModelConfig
        local depth = (math.abs(math.sin(math.rad(c.rotY or 0))) > 0.5) and bbSize.X or bbSize.Z
        local want = CFrame.new(c.offX or 0, c.offY or 0, (c.offZ or 0) - (depth/2 + (c.gap or 1.3)))
            * CFrame.Angles(math.rad(c.rotX or 0), math.rad(c.rotY or 0), math.rad(c.rotZ or 0))
        return want * pivotToBB:Inverse()
    end

    model:PivotTo(hrp.CFrame * computePivotInHrp())

    for _, pt in ipairs(parts) do
        if pt ~= primary then
            local w = Instance.new("Weld")
            w.Part0 = primary; w.Part1 = pt
            w.C0 = primary.CFrame:ToObjectSpace(pt.CFrame); w.C1 = CFrame.new()
            w.Parent = pt
        end
    end
    local hw = Instance.new("Weld")
    hw.Name = "MonoFlyWeld"
    hw.Part0 = hrp; hw.Part1 = primary
    hw.C0 = hrp.CFrame:ToObjectSpace(primary.CFrame); hw.C1 = CFrame.new()
    hw.Parent = primary
    for _, pt in ipairs(parts) do pt.Anchored = false end
    return model, {weld=hw, computePivotInHrp=computePivotInHrp}
end

local FXH = {}
do
    local NSK = NumberSequenceKeypoint.new
    local GRAV = Vector3.new(0, -130, 0)
    local bodies = {}
    local stepConn = nil
    local ROCK_MATS = {
        Enum.Material.Slate, Enum.Material.Basalt, Enum.Material.Granite,
        Enum.Material.Concrete, Enum.Material.Rock,
    }

    local function rayParams()
        local ex = {getFxFolder()}
        if p.Character then table.insert(ex, p.Character) end
        for _, gun in pairs(Guns) do if gun and gun.tool then table.insert(ex, gun.tool) end end
        local fm = workspace:FindFirstChild("MonoFlyModel"); if fm then table.insert(ex, fm) end
        local af = workspace:FindFirstChild("MonoAntiFling"); if af then table.insert(ex, af) end
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= p and pl.Character then table.insert(ex, pl.Character) end
        end
        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Exclude
        rp.FilterDescendantsInstances = ex
        rp.IgnoreWater = true
        return rp
    end
    FXH.rayParams = rayParams

    local function anchorAt(pos, life)
        local a = Instance.new("Part")
        a.Anchored = true; a.CanCollide = false; a.CanTouch = false; a.CanQuery = false
        a.CastShadow = false; a.Transparency = 1; a.Size = Vector3.new(0.2,0.2,0.2)
        a.CFrame = CFrame.new(pos); a.Parent = getFxFolder()
        if life then task.delay(life, function() if a then a:Destroy() end end) end
        return a
    end
    FXH.anchorAt = anchorAt

    local function lifeFade(part, life)
        task.delay(life, function()
            if part and part.Parent then
                TweenService:Create(part, TweenInfo.new(1, Enum.EasingStyle.Quad), {Transparency = 1}):Play()
                task.delay(1.05, function() if part then part:Destroy() end end)
            end
        end)
    end

    local function groundColor(hit)
        local c = Color3.fromRGB(125,122,118)
        local inst = hit and hit.Instance
        if inst and inst:IsA("BasePart") and not inst:IsA("Terrain") then
            c = c:Lerp(inst.Color, 0.55)
        end
        return c
    end

    local function step(dt)
        dt = math.min(dt, 0.05)
        if #bodies == 0 then
            if stepConn then stepConn:Disconnect(); stepConn = nil end
            return
        end
        local rp = rayParams()
        for i = #bodies, 1, -1 do
            local b = bodies[i]
            local part = b.part
            if not part or not part.Parent then
                table.remove(bodies, i)
            elseif not b.rest then
                b.vel = b.vel + GRAV*dt
                local sv = b.vel*dt
                local hit = nil
                if sv.Magnitude > 1e-4 then
                    hit = workspace:Raycast(b.pos, sv + sv.Unit*b.rad, rp)
                end
                if hit then
                    local n = hit.Normal
                    b.pos = hit.Position + n*(b.rad*0.9)
                    local vn = n*b.vel:Dot(n)
                    local vt = b.vel - vn
                    b.vel = vt*0.6 - vn*0.32
                    b.spin = b.spin*0.55
                    b.bounces = b.bounces + 1
                    if b.onImpact then
                        local f = b.onImpact; b.onImpact = nil
                        pcall(f, b, hit)
                    end
                    if b.vel.Magnitude < 8 or b.bounces >= 4 then
                        b.rest = true
                        local base = lookCF(b.pos, b.pos + n) * CFrame.Angles(-math.pi/2, 0, 0)
                        local restCF = base * CFrame.Angles(
                            (math.random()-0.5)*0.5, math.random()*6.283, (math.random()-0.5)*0.5)
                        part.CFrame = restCF + (-n*(b.rad*0.35))
                    else
                        b.rot = CFrame.Angles(b.spin.X*dt, b.spin.Y*dt, b.spin.Z*dt) * b.rot
                        part.CFrame = CFrame.new(b.pos) * b.rot
                    end
                else
                    b.pos = b.pos + sv
                    b.rot = CFrame.Angles(b.spin.X*dt, b.spin.Y*dt, b.spin.Z*dt) * b.rot
                    part.CFrame = CFrame.new(b.pos) * b.rot
                    if b.pos.Y < -500 then
                        table.remove(bodies, i); part:Destroy()
                    end
                end
            end
        end
    end

    function FXH.addBody(part, vel, spin, rad, life, onImpact)
        local cf = part.CFrame
        local b = {
            part = part, pos = cf.Position, rot = cf - cf.Position,
            vel = vel, spin = spin, rad = rad, bounces = 0, rest = false,
            onImpact = onImpact,
        }
        table.insert(bodies, b)
        if not stepConn then stepConn = RunService.Heartbeat:Connect(step) end
        lifeFade(part, life or 4)
    end

    function FXH.groundHit(pos, maxDist)
        local hit = workspace:Raycast(pos + Vector3.new(0,3,0),
            Vector3.new(0, -((maxDist or 20) + 3), 0), rayParams())
        if hit and hit.Normal.Y > 0.6 then return hit end
        return nil
    end

    function FXH.trail(part, dur, color, size)
        local e = Instance.new("ParticleEmitter")
        e.Rate = 40; e.Lifetime = NumberRange.new(0.5, 1.1)
        e.Speed = NumberRange.new(0.5, 3)
        e.SpreadAngle = Vector2.new(180, 180)
        e.Color = ColorSequence.new(color or Color3.fromRGB(80,76,72))
        e.Size = NumberSequence.new({NSK(0, size or 1), NSK(1, (size or 1)*3)})
        e.Transparency = NumberSequence.new({NSK(0, 0.5), NSK(1, 1)})
        e.LightInfluence = 1
        e.Parent = part
        task.delay(dur, function() if e and e.Parent then e.Enabled = false end end)
    end

    function FXH.shell(pos, color, material, startS, endS, dur, trStart, style)
        local s = Instance.new("Part")
        s.Shape = Enum.PartType.Ball; s.Anchored = true
        s.CanCollide = false; s.CanTouch = false; s.CanQuery = false; s.CastShadow = false
        s.Material = material or Enum.Material.Neon
        s.Color = color
        s.Transparency = trStart or 0
        s.Size = Vector3.new(startS, startS, startS)
        s.Position = pos
        s.Parent = getFxFolder()
        TweenService:Create(s, TweenInfo.new(dur, style or Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Size = Vector3.new(endS, endS, endS), Transparency = 1}):Play()
        task.delay(dur + 0.1, function() if s then s:Destroy() end end)
        return s
    end

    function FXH.light(pos, color, brightness, range, dur)
        local a = anchorAt(pos, dur + 0.3)
        local l = Instance.new("PointLight")
        l.Color = color; l.Brightness = brightness; l.Range = range; l.Shadows = false
        l.Parent = a
        TweenService:Create(l, TweenInfo.new(dur, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Brightness = 0, Range = range*0.4}):Play()
    end

    function FXH.sound(pos, id, vol, maxDist)
        local a = anchorAt(pos, 10)
        local s = Instance.new("Sound")
        s.SoundId = id; s.Volume = vol or 3
        s.RollOffMaxDistance = maxDist or 600
        s.RollOffMode = Enum.RollOffMode.InverseTapered
        s.Parent = a
        s:Play()
    end

    function FXH.shake(pos, intensity, dur, maxDist)
        local cam = workspace.CurrentCamera
        local ref = cam and cam.CFrame.Position
        if not ref then
            local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            ref = hrp and hrp.Position
        end
        if not ref then return end
        local k = math.clamp(1 - (ref - pos).Magnitude/(maxDist or 200), 0, 1)
        if k > 0.02 then cameraShake(intensity*k, dur) end
    end

    function FXH.ring(center, color, maxR, dur, height, segs)
        segs = segs or 32
        local fx = getFxFolder()
        local parts = {}
        for i = 1, segs do
            local pt = Instance.new("Part")
            pt.Anchored = true; pt.CanCollide = false; pt.CanTouch = false; pt.CanQuery = false
            pt.CastShadow = false; pt.Material = Enum.Material.Neon; pt.Color = color
            pt.Transparency = 0.2; pt.Size = Vector3.new(1, height, 1)
            pt.Parent = fx
            parts[i] = pt
        end
        task.spawn(function()
            local t0 = os.clock()
            while true do
                local u = (os.clock() - t0)/dur
                if u >= 1 then break end
                local e = 1 - (1-u)^3
                local r = 2 + (maxR - 2)*e
                local segLen = (2*math.pi*r/segs)*1.2
                local th = math.max(height*(1 - u*0.6), 0.05)
                local thick = math.max(0.5, 2.4*(1-u))
                local tr = 0.2 + 0.8*(u^1.5)
                for i = 1, segs do
                    local a = (i/segs)*math.pi*2
                    local dir = Vector3.new(math.cos(a), 0, math.sin(a))
                    local pos = center + dir*r + Vector3.new(0, th*0.5, 0)
                    local pt = parts[i]
                    pt.Size = Vector3.new(segLen, th, thick)
                    pt.CFrame = CFrame.lookAt(pos, pos + dir)
                    pt.Transparency = tr
                end
                RunService.Heartbeat:Wait()
            end
            for _, pt in ipairs(parts) do pt:Destroy() end
        end)
    end

        o = o or {}

        e.Lifetime = NumberRange.new(1.3, 2.6)

        e.Drag = 3
        e.Acceleration = Vector3.new(0, 2, 0)
        e.Color = ColorSequence.new(col)

        e.Rotation = NumberRange.new(0, 360)
        e.RotSpeed = NumberRange.new(-40, 40)
        e.LightInfluence = 1
        e.Parent = a

    end

    function FXH.spawnRocks(hit, o)
        o = o or {}
        local count = o.count or 22
        local power = o.power or 1
        local life = o.life or 4
        local sizeMul = o.sizeMul or 1
        local spread = o.spread or 1
        local gp = hit.Position
        local gcol = groundColor(hit)
        if o.greyer then
            local grey = (gcol.R + gcol.G + gcol.B) / 3
            gcol = Color3.new(grey, grey, grey):Lerp(Color3.fromRGB(160,158,155), 0.5)
        end
        local dustCol = Color3.fromRGB(95,90,84)
        local fx = getFxFolder()
        for _ = 1, count do
            local big = math.random() < 0.2
            local sz = (big and (1.3 + math.random()*1.5) or (0.35 + math.random()*0.85)) * sizeMul
            local part = Instance.new((math.random() < 0.3) and "WedgePart" or "Part")
            part.Name = "MonoRock"
            part.Anchored = true; part.CanCollide = false; part.CanTouch = false; part.CanQuery = false
            part.CastShadow = big
            part.Material = ROCK_MATS[math.random(1, #ROCK_MATS)]
            local shade = o.greyer and (1.00 + math.random()*0.35) or (0.72 + math.random()*0.5)
            part.Color = Color3.new(
                math.clamp(gcol.R*shade, 0, 1), math.clamp(gcol.G*shade, 0, 1), math.clamp(gcol.B*shade, 0, 1))
            part.Size = Vector3.new(sz, sz*(0.45 + math.random()*0.6), sz*(0.65 + math.random()*0.7))
            part.CFrame = CFrame.new(gp + Vector3.new(
                    (math.random()-0.5)*3*spread, 0.8 + math.random()*0.8, (math.random()-0.5)*3*spread))
                * CFrame.Angles(math.random()*6.283, math.random()*6.283, math.random()*6.283)
            part.Parent = fx
            local ang = math.random()*math.pi*2
            local h = (7 + math.random()*26) * power * (big and 0.7 or 1)
            local up = (26 + math.random()*38) * power * (big and 0.85 or 1)
            local vel = Vector3.new(math.cos(ang)*h, up, math.sin(ang)*h)
            local spin = Vector3.new((math.random()-0.5)*16, (math.random()-0.5)*16, (math.random()-0.5)*16)
            local rad = math.max(part.Size.X, part.Size.Y, part.Size.Z)*0.38
            FXH.addBody(part, vel, spin, rad, life + math.random()*0.6)
            if big then FXH.trail(part, 1.4, dustCol, sz*0.9) end
        end
    end

    function FXH.spawnPebbles(hit, radius, count, life, o)
        o = o or {}
        task.delay(0.1, function()
            local rp = rayParams()
            local gp = hit.Position
            local gcol = groundColor(hit)
            if o.greyer then
                local grey = (gcol.R + gcol.G + gcol.B) / 3
                gcol = Color3.new(grey, grey, grey):Lerp(Color3.fromRGB(155,153,150), 0.5)
            end
            local fx = getFxFolder()
            for _ = 1, count do
                local ang = math.random()*math.pi*2
                local r = radius*(0.2 + math.sqrt(math.random())*0.9)
                local from = gp + Vector3.new(math.cos(ang)*r, 4, math.sin(ang)*r)
                local h = workspace:Raycast(from, Vector3.new(0, -10, 0), rp)
                if h and h.Normal.Y > 0.6 then
                    local sz = 0.15 + math.random()*0.45
                    local pb = Instance.new("Part")
                    pb.Name = "MonoPebble"
                    pb.Anchored = true; pb.CanCollide = false; pb.CanTouch = false; pb.CanQuery = false
                    pb.CastShadow = false
                    pb.Material = ROCK_MATS[math.random(1, #ROCK_MATS)]
                    local shade = o.greyer and (1.00 + math.random()*0.35) or (0.7 + math.random()*0.5)
                    pb.Color = Color3.new(
                        math.clamp(gcol.R*shade, 0, 1), math.clamp(gcol.G*shade, 0, 1), math.clamp(gcol.B*shade, 0, 1))
                    pb.Size = Vector3.new(sz, sz*(0.5 + math.random()*0.5), sz*(0.7 + math.random()*0.6))
                    pb.CFrame = lookCF(h.Position, h.Position + h.Normal) * CFrame.Angles(-math.pi/2, 0, 0)
                        * CFrame.Angles((math.random()-0.5)*0.6, math.random()*6.283, (math.random()-0.5)*0.6)
                        + h.Normal*(sz*0.12)
                    pb.Parent = fx
                    lifeFade(pb, life + math.random()*0.8)
                end
            end
        end)
    end
end

        FXH.shell(pos + Vector3.new(0, 1.5, 0), Color3.fromRGB(255,75,18), Enum.Material.Neon, 2, 19, 0.55, 0.3)
    end)

            FXH.shell(pos + randUnit()*(3 + math.random()*5),
                (i % 2 == 0) and Color3.fromRGB(255,90,25) or Color3.fromRGB(255,195,90),

    fireP.Speed = NumberRange.new(18, 52)
    fireP.SpreadAngle = Vector2.new(180, 180)
    fireP.Drag = 3
    fireP.LightEmission = 1; fireP.LightInfluence = 0

    fireP.Parent = la

    local smoke = Instance.new("ParticleEmitter")
    smoke.Rate = 0; smoke.Lifetime = NumberRange.new(2.5, 4.5)
    smoke.Speed = NumberRange.new(6, 22)
    smoke.SpreadAngle = Vector2.new(180, 180)
    smoke.Acceleration = Vector3.new(0, 5, 0)
    smoke.Drag = 1.5
    smoke.LightInfluence = 1
    smoke.Rotation = NumberRange.new(0, 360)
    smoke.RotSpeed = NumberRange.new(-30, 30)
    smoke.Color = ColorSequence.new(Color3.fromRGB(62,58,54), Color3.fromRGB(24,24,24))
    smoke.Size = NumberSequence.new({NSK(0, 5), NSK(1, 21)})
    smoke.Transparency = NumberSequence.new({NSK(0, 0.35), NSK(0.5, 0.65), NSK(1, 1)})

    end

local function applyFlightPose(char)
    local J = getBodyJoints(char)
    local cfg = getgenv().FlyModelConfig
    local t = os.clock()
    local b = smooth01((t - flight.t0)/0.35)
    local sp = math.clamp(flight.vel.Magnitude / M.FLY.FAST, 0, 1)
    local lean = (cfg.lean + sp*12) * b
    local trem = math.sin(t*13) * (cfg.grip or 2.5) * (0.4 + sp)
    local sway = math.sin(t*2.2) * 1.2

    rot(J.root,  -lean, 0, 0)
    rot(J.waist, 6*b, 0, 0)
    rot(J.neck,  lean*0.9 + 6*b, 0, 0)

    rot(J.shoulderR, (cfg.armX + trem + sway)*b, 0, -cfg.armZ*b)
    rot(J.shoulderL, (cfg.armX - trem - sway)*b, 0,  cfg.armZ*b)
    rot(J.elbowR, (cfg.elbow - trem)*b, 0, 0)
    rot(J.elbowL, (cfg.elbow + trem)*b, 0, 0)
    rot(J.wristR, -22*b, 0, 0)
    rot(J.wristL, -22*b, 0, 0)

    rot(J.hipR, (cfg.legFwd + trem*0.7)*b, 0,  cfg.legSpread*b)
    rot(J.hipL, (cfg.legFwd - trem*0.7)*b, 0, -cfg.legSpread*b)
    rot(J.kneeR, -(cfg.kneeBend + trem)*b, 0, 0)
    rot(J.kneeL, -(cfg.kneeBend - trem)*b, 0, 0)
end

local function stopFlight(silent)
    if not flight.active and not flight.model then return end
    flight.active = false
    if flight.conn then pcall(function() flight.conn:Disconnect() end); flight.conn = nil end
    local char = p.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")

    local explodePos = nil
    if not silent and flight.model and flight.model.Parent then
        local okE, posE = pcall(flyExplosion, flight.model)
        if okE then explodePos = posE end
    end

    for _, k in ipairs({"lv","ao","att","model"}) do
        local o = flight[k]
        if o then pcall(function() o:Destroy() end); flight[k] = nil end
    end
    flight.weld = nil; flight.computePivotInHrp = nil
    if char then
        local J = getBodyJoints(char)
        for _, j in pairs(J) do resetJoint(j) end
    end
    if hum and hum.Parent then
        hum.PlatformStand = false
        hum.AutoRotate = true
        if hum.Health > 0 then pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
    end
    if hrp and hrp.Parent then
        local lv = hrp.CFrame.LookVector
        local flat = Vector3.new(lv.X, 0, lv.Z)
        if flat.Magnitude < 0.01 then flat = Vector3.new(0,0,-1) end
        hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + flat.Unit)
        hrp.AssemblyAngularVelocity = Vector3.zero
        if explodePos then
            local away = hrp.Position - explodePos
            if away.Magnitude < 0.1 then away = -flat end
            hrp.AssemblyLinearVelocity = away.Unit*26 + Vector3.new(0, 16, 0)
        end
    end
    flight.vel = Vector3.zero
    lastMoveStateAnim = nil
    if not silent then notify("Полёт", "ВЫКЛ [Q] · БАБАХ!") end
end
getgenv().hardStopFlight = function() stopFlight(true) end

local function startFlight()
    if flight.active then return end
    if getgenv().ShatterActive then return end
    local char = p.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return end
    if hum.SeatPart then hum.Sit = false end
    pcall(function() hum:UnequipTools() end)
    emoteToken = emoteToken + 1; emoteActive = false

    local model, meta = buildFlyModel(hrp)
    if not model then notify("Полёт", "Модель не загрузилась"); return end
    flight.model = model
    if meta then flight.weld = meta.weld; flight.computePivotInHrp = meta.computePivotInHrp end

    local att = Instance.new("Attachment"); att.Name = "MonoFlyAtt"; att.Parent = hrp
    local lv = Instance.new("LinearVelocity")
    lv.Attachment0 = att
    lv.RelativeTo = Enum.ActuatorRelativeTo.World
    lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    lv.MaxForce = math.huge
    lv.VectorVelocity = Vector3.zero
    lv.Parent = hrp
    local ao = Instance.new("AlignOrientation")
    ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
    ao.Attachment0 = att
    ao.RigidityEnabled = false
    ao.MaxTorque = math.huge
    ao.MaxAngularVelocity = math.huge
    ao.Responsiveness = 50
    local lk0 = hrp.CFrame.LookVector
    local f0 = Vector3.new(lk0.X, 0, lk0.Z)
    if f0.Magnitude < 0.01 then f0 = Vector3.new(0,0,-1) end
    ao.CFrame = CFrame.lookAt(Vector3.zero, f0.Unit)
    ao.Parent = hrp
    flight.att, flight.lv, flight.ao = att, lv, ao

    hum.PlatformStand = true
    hum.AutoRotate = false
    flight.vel = Vector3.zero
    flight.t0 = os.clock()
    flight.active = true
    lastMoveStateAnim = nil
    notify("Полёт", "ВКЛ · WASD · Space/Ctrl · Shift · [Q] выкл (взрыв)")

    flight.conn = RunService.Heartbeat:Connect(function(dt)
        if not flight.active then return end
        local ch = p.Character
        local h = ch and ch:FindFirstChildOfClass("Humanoid")
        local root = ch and ch:FindFirstChild("HumanoidRootPart")
        if not ch or not h or not root or not root.Parent or h.Health <= 0
            or not flight.model or not flight.model.Parent or not flight.lv or not flight.lv.Parent then
            stopFlight(true); return
        end
        if flight.weld and flight.weld.Parent and flight.computePivotInHrp then
            flight.weld.C0 = flight.computePivotInHrp()
        end
        local cam = workspace.CurrentCamera
        if not cam then return end
        local look = cam.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude < 0.01 then
            local rl = root.CFrame.LookVector
            flat = Vector3.new(rl.X, 0, rl.Z)
            if flat.Magnitude < 0.01 then flat = Vector3.new(0,0,-1) end
        end
        flat = flat.Unit
        local right = flat:Cross(Vector3.new(0,1,0))
        local f, s, v = 0, 0, 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then f = f + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then f = f - 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then s = s + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then s = s - 1 end
        if f == 0 and s == 0 then
            local md = h.MoveDirection
            if md.Magnitude > 0.1 then f = md:Dot(flat); s = md:Dot(right) end
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then v = v + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then v = v - 1 end
        local dir = look*f + right*s + Vector3.new(0, v, 0)
        if dir.Magnitude > 1 then dir = dir.Unit end
        local speed = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) and M.FLY.FAST or M.FLY.SPEED
        local vertOnly = Vector3.new(0, v*M.FLY.VERT, 0)
        local target = Vector3.new(dir.X*speed, 0, dir.Z*speed) + Vector3.new(0, dir.Y*speed, 0)
        if f == 0 and s == 0 then target = vertOnly end
        flight.vel = flight.vel:Lerp(target, math.clamp(dt*6, 0, 1))
        flight.lv.VectorVelocity = flight.vel
        if flight.ao and flight.ao.Parent then
            flight.ao.CFrame = CFrame.lookAt(Vector3.zero, flat)
        end
    end)
    track(flight.conn)
end

end

    -- БРОСОК / ПАДЕНИЕ
    local function spawnThrownBomb(startPos, velocity, opts)
        opts = opts or {}
        local model, parts = buildBombVisual()
        if not model then return nil end

        local _, bbSize = model:GetBoundingBox()
        local diameter = math.clamp(math.max(bbSize.X, bbSize.Y, bbSize.Z), 1.5, 8)
        local radius = diameter/2

        local ball = Instance.new("Part")
        ball.Name = "MonoBombPhysics"
        ball.Shape = Enum.PartType.Ball
        ball.Size = Vector3.new(diameter, diameter, diameter)
        ball.Transparency = 1
        ball.CanCollide = true; ball.CanTouch = false; ball.CanQuery = false
        ball.Anchored = true
        ball.CustomPhysicalProperties = PhysicalProperties.new(2.5, 0.9, 0, 100, 100)
        ball.CFrame = CFrame.new(startPos)
        ball.Parent = workspace

        for _, pt in ipairs(parts) do pt.Anchored = true end
        model.Parent = workspace
        local bbCF = model:GetBoundingBox()
        model:PivotTo(CFrame.new(startPos) * bbCF:ToObjectSpace(model:GetPivot()))
        for _, pt in ipairs(parts) do
            local w = Instance.new("Weld")
            w.Part0 = ball; w.Part1 = pt
            w.C0 = ball.CFrame:ToObjectSpace(pt.CFrame); w.C1 = CFrame.new()
            w.Parent = ball
        end
        for _, pt in ipairs(parts) do
            pt.Anchored = false; pt.CanCollide = false; pt.Massless = true
        end

        local beep = Instance.new("Sound")
        beep.Name = "MonoBombBeep"; beep.SoundId = M.SND.BOMB_BEEP
        beep.Volume = 2.5; beep.RollOffMaxDistance = 140
        beep.RollOffMode = Enum.RollOffMode.InverseTapered
        beep.Parent = ball

        ball.Anchored = false
        ball.AssemblyLinearVelocity = velocity
        local h0 = Vector3.new(velocity.X, 0, velocity.Z)
        ball.AssemblyAngularVelocity = Vector3.new(0, 1, 0):Cross(h0)/radius

        local fuse = opts.fuse or BOMB.fuse
        local handle = {
            pos = startPos, t = 0, fuse = fuse, grounded = false,
            exploded = false, explodePos = nil,
            dir = (h0.Magnitude > 1) and h0.Unit or nil,
        }

        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Exclude
        rp.FilterDescendantsInstances = {ball, model, getFxFolder()}
        rp.IgnoreWater = true

        local t0 = os.clock()
        local done = false
        local beepT = 0.5
        local hb
        hb = RunService.Heartbeat:Connect(function(dt)
            if done then return end
            if not ball.Parent then
                done = true; hb:Disconnect()
                if model then model:Destroy() end
                handle.exploded = true; handle.dead = true; handle.explodePos = handle.pos
                return
            end
            local t = os.clock() - t0
            handle.t = t
            handle.pos = ball.Position

            local v = ball.AssemblyLinearVelocity
            local ch = p.Character
            if ch then rp.FilterDescendantsInstances = {ball, model, getFxFolder(), ch} end
            local grounded = workspace:Raycast(ball.Position, Vector3.new(0, -(radius + 0.6), 0), rp) ~= nil
            handle.grounded = grounded
            if grounded then
                local h = Vector3.new(v.X, 0, v.Z)
                h = h*math.max(0, 1 - BOMB.rollDamping*dt)
                if h.Magnitude > BOMB.maxRollSpeed then h = h.Unit*BOMB.maxRollSpeed end
                ball.AssemblyLinearVelocity = Vector3.new(h.X, v.Y, h.Z)
                ball.AssemblyAngularVelocity = Vector3.new(0, 1, 0):Cross(h)/radius
            end

            beepT = beepT - dt
            if beepT <= 0 and t < fuse then
                pcall(function() beep.TimePosition = 0; beep:Play() end)
                local freq = math.clamp(t/fuse, 0, 1)
                beepT = math.max(0.1, 0.55 - 0.45*freq*freq)
            end

            if t >= fuse or ball.Position.Y < -500 then
                done = true; hb:Disconnect()
                local ep = ball.Position
                ball:Destroy()
                if model then model:Destroy() end
                handle.exploded = true
                handle.explodePos = ep
                pcall(bombExplosion, ep)
            end
        end)
        return handle
    end

    -- АНИМЕ-ПОРТАЛ
    local function animePortal(center, radius, groundY)
        local fx = getFxFolder()
        local P = {mode = "hidden", t0 = os.clock(), spin = 0, alive = true}
        local all = {}

        local function cyl(color, mat, tr)
            local c = Instance.new("Part")
            c.Shape = Enum.PartType.Cylinder
            c.Anchored = true; c.CanCollide = false; c.CanTouch = false; c.CanQuery = false; c.CastShadow = false
            c.Material = mat; c.Color = color; c.Transparency = tr
            c.Size = Vector3.new(0.1, 0.1, 0.1); c.CFrame = CFrame.new(center)
            c.Parent = fx
            table.insert(all, c)
            return c
        end

        local rim   = cyl(GREEN, Enum.Material.Neon, 0.05)
        local core  = cyl(GREEN_D, Enum.Material.SmoothPlastic, 0)
        local swirl = cyl(GREEN_L, Enum.Material.Neon, 0.5)
        local beamH = math.max(center.Y - (groundY or (center.Y - 30)), 4)
        local beam  = cyl(GREEN, Enum.Material.Neon, 0.88)

        local rings = {}
        for i = 1, 3 do
            local n = 18 + i*4
            local blocks = {}
            for j = 1, n do
                local b = Instance.new("Part")
                b.Anchored = true; b.CanCollide = false; b.CanTouch = false; b.CanQuery = false; b.CastShadow = false
                b.Material = Enum.Material.Neon
                b.Color = (j % 2 == 0) and GREEN or GREEN_L
                b.Size = Vector3.new(0.1, 0.1, 0.1)
                b.Parent = fx
                blocks[j] = b
                table.insert(all, b)
            end
            rings[i] = {blocks = blocks, n = n, r = radius*(1.12 + 0.16*i),
                dir = (i % 2 == 0) and 1 or -1, spd = 0.8 + 0.5*i}
        end

        local em = FXH.anchorAt(center, nil)
        table.insert(all, em)
        em.Size = Vector3.new(radius*2, 0.2, radius*2)
        local light = Instance.new("PointLight")
        light.Color = GREEN; light.Range = 55; light.Brightness = 0; light.Shadows = false
        light.Parent = em
        local spark = mkEmitter(em, {
            life = {0.6, 1.3}, speed = {14, 32}, spread = Vector2.new(22, 22), acc = Vector3.new(0, -18, 0), emit = 1,
            color = ColorSequence.new(WHITE, GREEN),
            size = NumberSequence.new({NSK(0, 0.7), NSK(1, 0)}),
            trans = NumberSequence.new({NSK(0, 0), NSK(1, 1)}),
        })
        spark.EmissionDirection = Enum.NormalId.Bottom
        pcall(function() spark.Shape = Enum.ParticleEmitterShape.Disc end)

        local base = CFrame.new(center) * CFrame.Angles(0, 0, math.pi/2)
        local function apply(s)
            s = math.max(s, 0.001)
            local d = radius*2*s
            rim.Size = Vector3.new(0.3, d*1.05, d*1.05); rim.CFrame = base
            core.Size = Vector3.new(0.2, d*0.9, d*0.9); core.CFrame = base * CFrame.new(-0.14, 0, 0)
            swirl.Size = Vector3.new(0.12, d*0.65, d*0.65); swirl.CFrame = base * CFrame.new(-0.26, 0, 0)
            beam.Size = Vector3.new(beamH, d*0.8, d*0.8)
            beam.CFrame = CFrame.new(center.X, center.Y - beamH/2, center.Z) * CFrame.Angles(0, 0, math.pi/2)
            for _, rg in ipairs(rings) do
                local r = rg.r*s
                local segLen = math.max((2*math.pi*r/rg.n)*0.6, 0.05)
                for j, b in ipairs(rg.blocks) do
                    local a = (j/rg.n)*math.pi*2 + P.spin*rg.dir*rg.spd
                    local pos = center + Vector3.new(math.cos(a)*r, math.sin(a*3 + P.spin*4)*0.35*s, math.sin(a)*r)
                    local tan = Vector3.new(-math.sin(a), 0, math.cos(a))
                    b.Size = Vector3.new(0.35*s + 0.01, 0.35*s + 0.01, segLen)
                    b.CFrame = CFrame.lookAt(pos, pos + tan)
                end
            end
        end

        function P.destroy()
            if not P.alive then return end
            P.alive = false
            if P.conn then P.conn:Disconnect() end
            for _, o in ipairs(all) do if o then o:Destroy() end end
        end
        function P.open()
            P.mode = "open"; P.t0 = os.clock()
            spark.Rate = 140
            TweenService:Create(light, TweenInfo.new(0.4), {Brightness = 5}):Play()
            FXH.sound(center, M.SND.SPAWN, 3, 500)
            FXH.ring(center, GREEN, radius*3, 0.55, 0.5, 40)
            pcall(function() spark:Emit(80) end)
        end
        function P.close()
            if P.mode == "close" or not P.alive then return end
            P.mode = "close"; P.t0 = os.clock()
            spark.Rate = 0
            TweenService:Create(light, TweenInfo.new(0.35), {Brightness = 0}):Play()
            FXH.sound(center, M.SND.ENTER, 2.5, 500)
            FXH.ring(center, GREEN_L, radius*2.2, 0.4, 0.4, 32)
        end

        apply(0)
        P.conn = RunService.RenderStepped:Connect(function(dt)
            if not P.alive then return end
            P.spin = P.spin + dt
            local el = os.clock() - P.t0
            local s = 0
            if P.mode == "open" then
                s = ease.back(el/0.45)
                if el >= 0.45 then P.mode = "idle" end
            elseif P.mode == "idle" then
                s = 1 + math.sin(P.spin*6)*0.02
            elseif P.mode == "close" then
                local u = el/0.4
                if u >= 1 then P.destroy(); return end
                s = 1 - ease.inOut(u)
            end
            apply(s)
        end)
        return P
    end

    -- ФАЗЫ КАМЕРЫ
    local function cineBombPhases(handle, ov, opts)
        opts = opts or {}
        local fl = CINE.cf.LookVector
        local flat = Vector3.new(fl.X, 0, fl.Z)
        local dir = opts.dir or handle.dir or ((flat.Magnitude > 0.01) and flat.Unit or Vector3.new(0, 0, -1))
        local rightV = dir:Cross(Vector3.new(0, 1, 0))
        local cur = CINE.cf

        local t0 = os.clock()
        while CINE.active and not CINE.skip and not handle.exploded do
            local dt = RunService.RenderStepped:Wait()
            local bp = handle.pos
            local desired = CFrame.lookAt(bp - dir*17 + rightV*6 + Vector3.new(0, 7, 0), bp)
            cur = cur:Lerp(desired, 1 - math.exp(-5*dt)); CINE.cf = cur
            -- без FOV-зума
            if opts.onTick then opts.onTick(bp) end
            local el = os.clock() - t0
            if (handle.grounded and el > 0.4) or el > 4 then break end
        end

        local ang = math.atan2(cur.Position.Z - handle.pos.Z, cur.Position.X - handle.pos.X)
        local lastShown = -1
        while CINE.active and not CINE.skip and not handle.exploded do
            local dt = RunService.RenderStepped:Wait()
            local rem = handle.fuse - handle.t
            local prog = math.clamp(handle.t/handle.fuse, 0, 1)
            ang = ang + dt*(0.7 + prog*1.4)
            local r = 16  -- радиус постоянный, без приближения
            local bp = handle.pos
            local desired = CFrame.lookAt(
                bp + Vector3.new(math.cos(ang)*r, 3.5 + 2*(1 - prog), math.sin(ang)*r),
                bp + Vector3.new(0, 1, 0))
            cur = cur:Lerp(desired, 1 - math.exp(-8*dt)); CINE.cf = cur
            -- без FOV-зума
            local n = math.ceil(rem)
            if n >= 1 and n <= 3 and n ~= lastShown then
                lastShown = n
                -- без текста: только shake
                CINE.shake(1.2 + (4 - n), 0.25)
            end
            if rem < 0.45 then CINE.shake(2.5, 0.1) end
        end

        if CINE.active then
            local ep = handle.explodePos or handle.pos
            -- без текста "БАБАХ!"
            if not CINE.skip then task.wait(0.12) end
            local startPos = CINE.cf.Position
            local baseAng = math.atan2(startPos.Z - ep.Z, startPos.X - ep.X)
            local c2 = CINE.cf
            CINE.run(4.6, function(u, dt)
                local e = ease.out(u)
                local r = 24 + 70*e
                local a = baseAng + u*0.9
                local h = 6 + 38*e
                local look = ep + Vector3.new(0, 4 + 46*e, 0)
                local desired = CFrame.lookAt(ep + Vector3.new(math.cos(a)*r, h, math.sin(a)*r), look)
                c2 = c2:Lerp(desired, 1 - math.exp(-4*dt)); CINE.cf = c2
                -- без FOV-зума
            end)
        end

        if CINE.active and CINE.saved then
            CINE.tween(CINE.saved.cf, CINE.saved.fov, 1.2, ease.inOut)
        end
    end

    local function endCinematic(ov)
        pcall(function() ov.bars(false, 0.4) end)
        task.wait(0.4)
        pcall(function() ov.destroy() end)
        CINE.finish()
    end

    local function runPortalSequence(target)
        if CINE.active or busy then return end
        busy = true
        local ov = makeOverlay()
        CINE.begin()
        local portal
        local ok, err = pcall(function()
            local H = BOMB.portalHeight
            local portalPos = target + Vector3.new(0, H, 0)
            local camPos = CINE.cf.Position
            local back = Vector3.new(camPos.X - target.X, 0, camPos.Z - target.Z)
            if back.Magnitude < 5 then back = Vector3.new(0, 0, 1) end
            back = back.Unit
            local right = back:Cross(Vector3.new(0, 1, 0))

            ov.bars(true, 0.4)
            ov.lines(0.7, Color3.new(1, 1, 1))
            local shot1 = CFrame.lookAt(
                target + back*58 + right*14 + Vector3.new(0, H*0.5, 0),
                target + Vector3.new(0, H*0.62, 0))
            CINE.tween(shot1, CINE.fov, 0.95, ease.inOut) -- FOV не меняется

            portal = animePortal(portalPos, 9, target.Y)
            portal.open()
            ov.flash(GREEN_L, 0.55, 0.35)
            CINE.shake(5, 0.5)
            -- долли-ин удалён: камера не приближается

            local handle = spawnThrownBomb(portalPos - Vector3.new(0, 4, 0), Vector3.new(0, -26, 0),
                {fuse = BOMB.dropFuse})
            if not handle then error("модель бомбы не загрузилась") end

            local closed = false
            cineBombPhases(handle, ov, {
                dir = -back,
                onTick = function(bp)
                    if not closed and bp.Y < portalPos.Y - 10 then
                        closed = true; portal.close()
                    end
                end,
            })
        end)
        if not ok then warn("[MONO] cine:", err) end
        if portal then portal.destroy() end
        endCinematic(ov)
        busy = false
    end

        endCinematic(ov)
        busy = false
    end

        if CINE.active or busy then return end

            local ray = cam:ViewportPointToRay(mp.X, mp.Y)
            local res = workspace:Raycast(ray.Origin, ray.Direction*2000, FXH.rayParams())

                marker.CFrame = CFrame.new(res.Position + Vector3.new(0, 0.15, 0)) * CFrame.Angles(0, 0, math.pi/2)
                marker.Size = Vector3.new(0.2, 8 + math.sin(os.clock()*6)*0.8, 8 + math.sin(os.clock()*6)*0.8)

            if gp then return end
            if input.UserInputType ~= Enum.UserInputType.MouseButton1
                and input.UserInputType ~= Enum.UserInputType.Touch then return end
            if not lastHit then notify("Бомба", "Нужна ровная поверхность"); return end
            local target = lastHit.Position
            stopPick()
            task.spawn(runPortalSequence, target)
        end)
    end

local function stopToolAnims(hum)
    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then return end
    for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
        local nm = string.lower((tr.Animation and tr.Animation.Name) or tr.Name or "")
        if string.find(nm, "tool", 1, true) then
            pcall(function() tr:Stop(0) end)
        end
    end
end

local gunArmActive = false
local function applyGunArm(char, hum, gun)
    stopToolAnims(hum)
    local J = getBodyJoints(char)
    local raise = gun.def.noShoot and 0 or gunRaiseAmount()
    if raise > 0.001 then
        if J.shoulderR then
            local cam = workspace.CurrentCamera
            local pitch = cam and math.deg(math.asin(math.clamp(cam.CFrame.LookVector.Y, -1, 1))) or 0
            local ang = math.clamp(88 + pitch, 20, 170)
            applyJointRot(J.shoulderR, rotCache, ang*raise, 0, 0)
        end
        resetJoint(J.elbowR)
        resetJoint(J.wristR)
        gunArmActive = true
    elseif gunArmActive then
        resetJoint(J.shoulderR)
        resetJoint(J.elbowR)
        resetJoint(J.wristR)
        gunArmActive = false
    end
end

end

            if pl ~= p and pl.Character then

    local function check()
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hum or not hrp or hum.Health <= 0 then
            AF.lastSafeCF = nil; AF.lastPos = nil; return
        end
        local now = os.clock()
        if flight.active then AF.wasFlying = true
        elseif AF.wasFlying then
            AF.wasFlying = false
            AF.suspendUntil = math.max(AF.suspendUntil, now + 1.5)
        end
        local exempt = flight.active or getgenv().ShatterActive or isTeleporting
            or now < AF.suspendUntil or now - lastTeleport < 1.0
            or hum.SeatPart ~= nil or hrp.Anchored
        if exempt then
            AF.lastSafeCF = hrp.CFrame; AF.lastPos = hrp.Position; return
        end

        local lv = hrp.AssemblyLinearVelocity
        local av = hrp.AssemblyAngularVelocity
        local hLim, uLim, aLim = limits()
        local horiz = Vector3.new(lv.X, 0, lv.Z).Magnitude
        local bad = horiz > hLim or lv.Y > uLim or av.Magnitude > aLim

        local jumped = false
        if AF.lastPos and (hrp.Position - AF.lastPos).Magnitude > 60
            and (bad or now - AF.abnormalAt < 0.6 or nearOther(30)) then
            jumped = true
        end

        if bad then
            AF.abnormalAt = now
            hrp.AssemblyLinearVelocity = Vector3.new(0, math.clamp(lv.Y, -80, 30), 0)
            hrp.AssemblyAngularVelocity = Vector3.zero
        end

        if jumped and AF.lastSafeCF and now >= AF.blockUntil then
            local fresh = {}
            for _, t in ipairs(AF.rollbacks) do
                if now - t < 2 then table.insert(fresh, t) end
            end
            table.insert(fresh, now)
            AF.rollbacks = fresh
            if #fresh >= 4 then
                AF.blockUntil = now + 3
            else
                hrp.CFrame = AF.lastSafeCF
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end
        end

        if bad or jumped then
            trigger()
            AF.lastPos = hrp.Position
        else
            AF.lastSafeCF = hrp.CFrame
            AF.lastPos = hrp.Position
        end
    end

local scannerRayParams = RaycastParams.new()
scannerRayParams.FilterType=Enum.RaycastFilterType.Exclude

local function findPlayerByLook()
    local camCF = Camera.CFrame
    local origin = camCF.Position
    local lookDir = camCF.LookVector
    local myChar = LocalPlayer.Character
    if stickyTarget and stickyTarget.Parent and stickyTarget.Character then
        local head = stickyTarget.Character:FindFirstChild("Head")
        if head then
            local to = head.Position - origin
            local d = to.Magnitude
            if d <= M.SC.STICKY_DIST and d > 0.1 then
                local dot = lookDir:Dot(to.Unit)
                local ang = math.deg(math.acos(math.clamp(dot,-1,1)))
                if ang < M.SC.STICKY_ANGLE then
                    scannerRayParams.FilterDescendantsInstances = {myChar, stickyTarget.Character}
                    if not workspace:Raycast(origin, to, scannerRayParams) then return stickyTarget, head end
                end
            end
        end
        stickyTarget = nil
    end
    local bestP, bestH, bestA = nil, nil, M.SC.LOOK_ANGLE
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and pl.Character then
            local head = pl.Character:FindFirstChild("Head")
            if head then
                local to = head.Position - origin
                local d = to.Magnitude
                if d <= M.SC.LOOK_DIST and d > 0.1 then
                    local dot = lookDir:Dot(to.Unit)
                    local ang = math.deg(math.acos(math.clamp(dot,-1,1)))
                    if ang < bestA then
                        scannerRayParams.FilterDescendantsInstances = {myChar, pl.Character}
                        if not workspace:Raycast(origin, to, scannerRayParams) then
                            bestA = ang; bestP = pl; bestH = head
                        end
                    end
                end
            end
        end
    end
    if bestP then stickyTarget = bestP end
    return bestP, bestH
end

-- HOLE
local function portalRadius() return (GUN_PORTAL_STYLE.portalSize or M.DM.PORTAL_SIZE)*0.45 end
local function holeDisable(part)
    local reg = getgenv().MonoHoleParts
    local n = reg[part] or 0
    if n == 0 then part.CanCollide = false end
    reg[part] = n+1
end
local function holeRestore(part)
    local reg = getgenv().MonoHoleParts
    local n = reg[part]; if not n then return end
    n = n - 1
    if n <= 0 then
        reg[part] = nil
        if part and part.Parent then pcall(function() part.CanCollide = true end) end
    else reg[part] = n end
end
local function isFaceAligned(part, point, norm)
    local pcf = part.CFrame
    local rel = pcf:PointToObjectSpace(point)
    local sz = part.Size
    local axes = {{pcf.RightVector,rel.X,sz.X},{pcf.UpVector,rel.Y,sz.Y},{pcf.LookVector,rel.Z,sz.Z}}
    for _, a in ipairs(axes) do
        if math.abs(a[1]:Dot(norm)) > 0.9 and math.abs(math.abs(a[2]) - a[3]/2) < 0.9 then return true end
    end
    return false
end
local function activateHole(po)
    if po.holeActive then return end
    local base = po.part; if not base or not base.Parent then return end
    po.holeActive = true; po.disabled = po.disabled or {}
    local S = GUN_PORTAL_STYLE.portalSize or M.DM.PORTAL_SIZE
    local ex = {base}
    if p.Character then table.insert(ex, p.Character) end
    local op = OverlapParams.new()
    op.FilterType = Enum.RaycastFilterType.Exclude
    op.FilterDescendantsInstances = ex
    local ok, parts = pcall(function()
        return workspace:GetPartBoundsInBox(base.CFrame, Vector3.new(S*1.1,S*1.1,3), op)
    end)
    if not ok or not parts then parts = {} end
    if po.hitPart and po.hitPart.Parent then table.insert(parts, po.hitPart) end
    for _, part in ipairs(parts) do
        if part:IsA("BasePart") and not part:IsA("Terrain") and part.Anchored and part.CanCollide
            and not po.disabled[part] then
            if part == po.hitPart or isFaceAligned(part, po.position, po.normal) then
                po.disabled[part] = true; holeDisable(part)
            end
        end
    end
end
local function releaseHole(po, pushCF)
    if po.disabled then
        for part in pairs(po.disabled) do holeRestore(part) end
        po.disabled = {}
    end
    po.holeActive = false
    if pushCF then
        local char = p.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            local rel = pushCF:PointToObjectSpace(root.Position)
            local lat = math.sqrt(rel.X*rel.X + rel.Y*rel.Y)
            if rel.Z > -0.6 and rel.Z < 6 and lat < portalRadius()+1 then
                local norm = -pushCF.LookVector
                root.CFrame = root.CFrame + (-norm)*(3 - rel.Z)
                root.AssemblyLinearVelocity = Vector3.zero
            end
        end
    end
end

destroyPortalPart = function(po)
    if not po or not po.part or not po.part.Parent then return end
    local base = po.part; local baseCF = base.CFrame
    po.part = nil; po.isDestroying = true
    local dt = GUN_PORTAL_STYLE.disappearTime or 0.30
    if po.model then
        po.destroyT0 = os.clock(); po.destroyDur = dt
        releaseHole(po, baseCF)
        task.delay(dt + 0.1, function() if base and base.Parent then base:Destroy() end end)
        return
    end
    if po.spinConnection then po.spinConnection:Disconnect(); po.spinConnection = nil end
    releaseHole(po, baseCF)
    local surf = base:FindFirstChild("PortalSurface")
    if surf then
        for _, sg in ipairs(surf:GetChildren()) do
            if sg:IsA("SurfaceGui") then
                local img = sg:FindFirstChild("PortalImage")
                if img then
                    TweenService:Create(img, TweenInfo.new(dt, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
                        {ImageTransparency=1}):Play()
                end
            end
        end
        local sz = surf.Size
        TweenService:Create(surf, TweenInfo.new(dt, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            {Size=Vector3.new(sz.X*0.06, sz.Y*1.05, 0.04)}):Play()
    end
    task.delay(dt+0.1, function() if base and base.Parent then base:Destroy() end end)
end

local function createRickPortal(pos, norm, isReturn, texture, hitPart)
    texture = texture or "rbxassetid://129332930839385"
    local PS = GUN_PORTAL_STYLE
    sSnd:Play(); cameraShake(0.15, 0.25)
    local S = PS.portalSize or M.DM.PORTAL_SIZE
    local center = pos + norm*M.DM.PORTAL_GAP
    local up = Vector3.new(0,1,0)
    if math.abs(norm.Y) > 0.99 then up = Vector3.new(0,0,1) end
    local base = Instance.new("Part")
    base.Name="PortalBase"; base.Size=Vector3.new(0.6,0.6,0.6); base.Transparency=1
    base.Anchored=true; base.CanCollide=false; base.CanTouch=false; base.CanQuery=false; base.CastShadow=false
    base.CFrame=CFrame.lookAt(center, center+norm, up); base.Parent=workspace
    local slit = Vector3.new(S*PS.openSlitW, S*PS.openSlitH, 0.06)
    local surf = Instance.new("Part")
    surf.Name="PortalSurface"; surf.Transparency=1; surf.Anchored=true
    surf.CanCollide=false; surf.CanTouch=false; surf.CanQuery=false; surf.CastShadow=false
    surf.Size = slit; surf.CFrame = base.CFrame; surf.Parent = base
    local imgT = PS.imageFinalTrans or 0
    for _, face in ipairs({Enum.NormalId.Front, Enum.NormalId.Back}) do
        local sg = Instance.new("SurfaceGui")
        sg.Name = face==Enum.NormalId.Front and "PortalGuiFront" or "PortalGuiBack"
        sg.Face = face; sg.LightInfluence=0; sg.AlwaysOnTop=false
        sg.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
        sg.PixelsPerStud = 100; sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        sg.Parent = surf
        local img = Instance.new("ImageLabel")
        img.Name="PortalImage"; img.Size=UDim2.fromScale(1,1); img.Position=UDim2.fromScale(0,0)
        img.BackgroundTransparency=1; img.Image=texture
        img.ImageColor3=Color3.fromRGB(255,255,255); img.ImageTransparency=1
        img.ScaleType=Enum.ScaleType.Stretch; img.ZIndex=1; img.Parent=sg
    end
    local po = {
        type="Rick", part=base, position=center, normal=norm,
        isReturnPortal=isReturn or false, isDestroying=false,
        hitPart=hitPart, disabled={}, holeActive=false,
        prevZ=nil, justArrived=false, wasInsideH=false,
    }
    local t1,t2,t3 = PS.openT1,PS.openT2,PS.openT3
    local appearAt = os.clock() + t1+t2+t3
    local wide = Vector3.new(S*PS.openWideW, S*PS.openWideH, 0.06)
    local settle = Vector3.new(S*PS.openSettleW, S*PS.openSettleH, 0.06)
    local final = Vector3.new(S, S, 0.06)
    for _, sg in ipairs(surf:GetChildren()) do
        if sg:IsA("SurfaceGui") then
            local img = sg:FindFirstChild("PortalImage")
            if img then
                TweenService:Create(img, TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                    {ImageTransparency=imgT}):Play()
            end
        end
    end
    task.spawn(function()
        local function step(sz, dur, style, dir)
            if po.isDestroying or not surf.Parent then return false end
            local tw = TweenService:Create(surf, TweenInfo.new(dur, style, dir), {Size=sz})
            tw:Play(); tw.Completed:Wait()
            return not po.isDestroying and surf.Parent ~= nil
        end
        if not step(wide, t1, Enum.EasingStyle.Back, Enum.EasingDirection.Out) then return end
        if not step(settle, t2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut) then return end
        step(final, t3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    end)
    local rotA = 0
    local spin
    spin = RunService.RenderStepped:Connect(function(dt)
        if not base.Parent then spin:Disconnect(); return end
        if po.isDestroying or os.clock() < appearAt then return end
        rotA = rotA + dt*(PS.spinSpeed or 0.65)
        surf.CFrame = base.CFrame * CFrame.Angles(0,0,rotA)
    end)
    po.spinConnection = spin
    task.spawn(function()
        task.wait(M.DM.PORTAL_LIFE)
        if portal1 == po then destroyPortalPart(portal1); portal1=nil; arrivedAt=nil
        elseif portal2 == po then destroyPortalPart(portal2); portal2=nil; arrivedAt=nil end
    end)
    return po
end

getgenv().PortalModelConfig = {
    autoAlign = true,
    scale = 1.3,
    offX = 0, offY = 0, offZ = 0,
    rotX = 0, rotY = 0, rotZ = 0,
    spinSpeed = 0.65,
}
do
    local CACHE = {}
    local function getSrc(id)
        if not id then return false end
        if CACHE[id] ~= nil then return CACHE[id] end
        local ok, m = pcall(loadAssetModel, id)
        CACHE[id] = (ok and m) or false
        return CACHE[id]
    end
    EXT.portalPreload = function(id) getSrc(id) end

    local function easeOutBack(u)
        local c1 = 1.70158; local c3 = c1 + 1
        return 1 + c3*(u-1)^3 + c1*(u-1)^2
    end

    EXT.createModelPortal = function(pos, norm, isReturn, modelId, hitPart)
        local src = getSrc(modelId)
        if not src then return nil end
        local PS = GUN_PORTAL_STYLE
        local cfg = getgenv().PortalModelConfig
        local S = PS.portalSize or M.DM.PORTAL_SIZE

        local model = src:Clone()
        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript") then d:Destroy() end
        end
        local nParts = 0
        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("BasePart") then
                nParts = nParts + 1
                d.Anchored = true; d.CanCollide = false; d.CanTouch = false
                d.CanQuery = false; d.CastShadow = false
            end
        end
        if nParts == 0 then model:Destroy(); return nil end
        pcall(function() model.PrimaryPart = nil end)

        local bbCF0, bbSize0 = model:GetBoundingBox()
        local alignRot = CFrame.new()
        local planar = math.max(bbSize0.X, bbSize0.Y, bbSize0.Z)
        if cfg.autoAlign then
            if bbSize0.X <= bbSize0.Y and bbSize0.X <= bbSize0.Z then
                alignRot = CFrame.Angles(0, math.pi/2, 0)
                planar = math.max(bbSize0.Y, bbSize0.Z)
            elseif bbSize0.Y <= bbSize0.Z then
                alignRot = CFrame.Angles(math.pi/2, 0, 0)
                planar = math.max(bbSize0.X, bbSize0.Z)
            else
                planar = math.max(bbSize0.X, bbSize0.Y)
            end
        end
        if planar < 0.01 then planar = 1 end
        pcall(function() model:ScaleTo(model:GetScale() * (S / planar) * (cfg.scale or 1)) end)
        local scale0 = model:GetScale()
        local bbCF = model:GetBoundingBox()
        local rel = bbCF:ToObjectSpace(model:GetPivot())
        local relRot = rel - rel.Position
        local relPos = rel.Position
        local userCF = CFrame.new(cfg.offX or 0, cfg.offY or 0, cfg.offZ or 0)
            * CFrame.Angles(math.rad(cfg.rotX or 0), math.rad(cfg.rotY or 0), math.rad(cfg.rotZ or 0))

        sSnd:Play(); cameraShake(0.15, 0.25)
        local center = pos + norm*M.DM.PORTAL_GAP
        local up = Vector3.new(0,1,0)
        if math.abs(norm.Y) > 0.99 then up = Vector3.new(0,0,1) end
        local base = Instance.new("Part")
        base.Name = "PortalBase"; base.Size = Vector3.new(0.6,0.6,0.6); base.Transparency = 1
        base.Anchored = true; base.CanCollide = false; base.CanTouch = false
        base.CanQuery = false; base.CastShadow = false
        base.CFrame = CFrame.lookAt(center, center+norm, up)
        base.Parent = workspace
        model.Name = "PortalModel"
        model.Parent = base

        local po = {
            type = "RickModel", part = base, position = center, normal = norm,
            isReturnPortal = isReturn or false, isDestroying = false,
            hitPart = hitPart, disabled = {}, holeActive = false,
            prevZ = nil, justArrived = false, wasInsideH = false,
            model = model, pulse = nil,
        }

        local openDur = (PS.openT1 or 0.11) + (PS.openT2 or 0.12) + (PS.openT3 or 0.14)
        local openT0 = os.clock()
        local rotA, lastM = 0, -1
        local spin
        spin = RunService.RenderStepped:Connect(function(dt)
            if not base.Parent or not model.Parent then spin:Disconnect(); return end
            local now = os.clock()
            local u = math.clamp((now - openT0)/openDur, 0, 1)
            local m = 0.04 + 0.96*easeOutBack(u)
            if po.pulse then
                local pe = (now - po.pulse.t0)/0.43
                if pe >= 1 then po.pulse = nil
                else m = m*(1 + po.pulse.amp*math.sin(math.pi*pe)) end
            end
            if po.isDestroying then
                local k = math.clamp((now - (po.destroyT0 or now))/(po.destroyDur or 0.3), 0, 1)
                m = m*math.max(1 - k*k, 0.01)
            elseif u >= 1 then
                rotA = rotA + dt*(cfg.spinSpeed or 0)
            end
            if math.abs(m - lastM) > 0.002 then
                lastM = m
                pcall(function() model:ScaleTo(scale0*m) end)
            end
            local relM = CFrame.new(relPos*m) * relRot
            local targetCF = base.CFrame * CFrame.Angles(0,0,rotA) * userCF * alignRot * relM
            pcall(function()
                model:PivotTo(targetCF)
            end)
            if po.isDestroying and (now - (po.destroyT0 or now)) > (po.destroyDur or 0.3) + 0.05 then
                spin:Disconnect()
            end
        end)
        po.spinConnection = spin

        task.spawn(function()
            task.wait(M.DM.PORTAL_LIFE)
            if portal1 == po then destroyPortalPart(portal1); portal1=nil; arrivedAt=nil
            elseif portal2 == po then destroyPortalPart(portal2); portal2=nil; arrivedAt=nil end
        end)
        return po
    end
end

local function createPortalPart(pos, norm, isReturn, texture, hitPart, modelId)
    if modelId and EXT.createModelPortal then
        local ok, po = pcall(EXT.createModelPortal, pos, norm, isReturn, modelId, hitPart)
        if ok and po then return po end
        if not ok then warn("[MONO] Портал-модель: " .. tostring(po) .. " -> fallback на картинку") end
    end
    return createRickPortal(pos, norm, isReturn, texture, hitPart)
end

local function loadMapWithSpawn(assetId, name, center, scale, offY, spName, snap)
    offY = offY or 5; spName = spName or "SpawnLocation"
    local model = loadAssetModel(assetId)
    if not model then return nil, nil end
    if not model:IsA("Model") then local w = Instance.new("Model"); model.Parent=w; model=w end
    model.Name = name; model.Parent = workspace
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript") then d:Destroy() end
    end
    positionModel(model, center)
    pcall(function() model:ScaleTo(scale) end)
    for _, part in ipairs(model:GetDescendants()) do
        if part:IsA("BasePart") then
            part.Anchored = true
            if part.Transparency >= 0.9 and part.CanCollide then part.CanCollide = false end
        end
    end
    local sp, isReal = nil, false
    if spName ~= "SpawnLocation" then
        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("BasePart") and d.Name == spName then sp = d; isReal = d:IsA("SpawnLocation"); break end
        end
    end
    if not sp then
        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("SpawnLocation") then sp = d; isReal = true; break end
        end
    end
    if not sp then
        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("BasePart") and d.Name == spName then sp = d; break end
        end
    end
    local spPoint
    if sp then
        local bp = sp.Position
        if snap and not isReal then
            local rp = RaycastParams.new()
            rp.FilterType = Enum.RaycastFilterType.Exclude
            local ex = {}
            if p.Character then table.insert(ex, p.Character) end
            for _, g in pairs(Guns) do if g and g.tool then table.insert(ex, g.tool) end end
            if portal1 and portal1.part then table.insert(ex, portal1.part) end
            if portal2 and portal2.part then table.insert(ex, portal2.part) end
            table.insert(ex, sp); rp.FilterDescendantsInstances = ex
            local hit = workspace:Raycast(bp+Vector3.new(0,2,0), Vector3.new(0,-1500,0), rp)
            if hit then bp = hit.Position end
        end
        spPoint = bp + Vector3.new(0, offY, 0)
    else
        local best, bestS = nil, -math.huge
        for _, part in ipairs(model:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                local sz = part.Size
                local h = math.min(sz.X, sz.Z)
                if h >= 8 then
                    local s = part.Position.Y + h*0.4
                    if s > bestS then bestS = s; best = part end
                end
            end
        end
        if best then spPoint = best.Position + Vector3.new(0, best.Size.Y/2 + offY, 0)
        else spPoint = center + Vector3.new(0,10,0) end
    end
    return model, spPoint
end

-- PORTAL LOGIC
local function getPortalCF(po)
    if not po or not po.part or not po.part.Parent then return nil end
    if po.part:IsA("Model") then return po.part:GetPivot() end
    return po.part.CFrame
end
local function pointInside(po, pt)
    local cf = getPortalCF(po); if not cf then return false end
    local rel = cf:PointToObjectSpace(pt)
    if rel.Z > 2.2 or rel.Z < -1.8 then return false end
    return (rel.X*rel.X + rel.Y*rel.Y) <= 12.25
end
local function insidePortalH(po, root)
    if not po or not po.part or not po.part.Parent then return false end
    local bp = root.Position
    return pointInside(po, bp)
        or pointInside(po, bp - Vector3.new(0,2.0,0))
        or pointInside(po, bp + Vector3.new(0,1.2,0))
end
local function animatePortalPulse(po, scale)
    if not po or not po.part or not po.part.Parent then return end
    if po.model then po.pulse = {t0=os.clock(), amp=scale-1}; return end
    local s = po.part:FindFirstChild("PortalSurface"); if not s then return end
    local bs = s.Size
    local big = Vector3.new(bs.X*scale, bs.Y*scale, bs.Z)
    local tw = TweenService:Create(s, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size=big})
    tw:Play()
    tw.Completed:Connect(function()
        if s and s.Parent then
            TweenService:Create(s, TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Size=bs}):Play()
        end
    end)
end
local function horizontalTeleport(from, to)
    local char = p.Character
    if not char or not to or not to.part or not to.part.Parent then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return end
    local now = os.clock()
    if now - lastTeleport < M.DM.COOLDOWN or isTeleporting then return end
    isTeleporting = true; lastTeleport = now
    if hum.SeatPart then hum.Sit = false end
    animatePortalPulse(from, 1.15)
    tSnd:Play(); if enterSnd then enterSnd:Play() end
    local eP, eN = from.position, from.normal
    local xP, xN = to.position, to.normal
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.Anchored = true
    local eLook = Vector3.new(-eN.X, 0, -eN.Z)
    if eLook.Magnitude < 0.01 then eLook = hrp.CFrame.LookVector else eLook = eLook.Unit end
    local pullCF = CFrame.lookAt(eP - eN*0.8, eP - eN*0.8 + eLook)
    local pt = TweenService:Create(hrp, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {CFrame=pullCF})
    pt:Play(); pt.Completed:Wait()
    local vel, ecf
    local flat = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
    if flat.Magnitude < 0.01 then flat = Vector3.new(0,0,-1) else flat = flat.Unit end
    if xN.Y > 0.7 then
        ecf = CFrame.lookAt(xP + Vector3.new(0,3.2,0), xP + Vector3.new(0,3.2,0) + flat)
        vel = Vector3.new(0,3,0)
    elseif xN.Y < -0.7 then
        ecf = CFrame.lookAt(xP - Vector3.new(0,3.5,0), xP - Vector3.new(0,3.5,0) + flat)
        vel = flat*4
    else
        local wd = Vector3.new(xN.X, 0, xN.Z)
        if wd.Magnitude < 0.01 then wd = hrp.CFrame.LookVector else wd = wd.Unit end
        ecf = CFrame.lookAt(xP + xN*3.0, xP + xN*3.0 + wd)
        vel = wd*6
    end
    hrp.CFrame = ecf; hrp.Anchored = false
    hrp.AssemblyLinearVelocity = vel
    arrivedAt = to; isTeleporting = false
    to.wasInsideH = true
    if to.isReturnPortal then
        returnPosition = nil; currentDestination = "Normal"
        notify("Возврат", "Вы на Земле!")
    end
end
local function yawOf(v) return math.atan2(-v.X, -v.Z) end

    tSnd:Play(); if enterSnd then enterSnd:Play() end
    local A = fromP.part.CFrame
    local B = toP.part.CFrame
    local eN, xN = fromP.normal, toP.normal
    local rel = A:PointToObjectSpace(hrp.Position)
    local lz = math.min(-rel.Z, -2.2)
    local nP = B:PointToWorldSpace(Vector3.new(-rel.X, rel.Y, lz))
    local mD = 2.2
    if xN.Y > 0.7 then mD = 3.4 elseif xN.Y < -0.7 then mD = 3.6 end
    local d = (nP - toP.position):Dot(xN)
    if d < mD then nP = nP + xN*(mD-d) end
    local T = B * CFrame.Angles(0, math.pi, 0)
    local nV = T:VectorToWorldSpace(A:VectorToObjectSpace(hrp.AssemblyLinearVelocity))
    local mS = 4
    if xN.Y > 0.7 then mS = 5 end
    local al = nV:Dot(xN)
    if al < mS then nV = nV + xN*(mS-al) end
    local oldL = hrp.CFrame.LookVector
    local oldF = Vector3.new(oldL.X, 0, oldL.Z)
    if oldF.Magnitude < 0.01 then oldF = Vector3.new(0,0,-1) end
    oldF = oldF.Unit
    local mL = T:VectorToWorldSpace(A:VectorToObjectSpace(oldL))
    local mF = Vector3.new(mL.X, 0, mL.Z)
    local face
    if math.abs(eN.Y)>0.7 and math.abs(xN.Y)<0.7 then face = Vector3.new(xN.X, 0, xN.Z)
    elseif mF.Magnitude > 0.3 then face = mF
    else face = oldF end
    if face.Magnitude < 0.01 then face = oldF end
    face = face.Unit
    local dY = yawOf(face) - yawOf(oldF)
    local camCF = Camera.CFrame
    local oldPos = hrp.Position
    local camRel = CFrame.new(-oldPos) * camCF
    hrp.CFrame = CFrame.lookAt(nP, nP + face)
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.AssemblyLinearVelocity = nV
    pcall(function() Camera.CFrame = CFrame.new(nP) * CFrame.Angles(0,dY,0) * camRel end)
    fromP.prevZ = nil; toP.prevZ = nil
    toP.justArrived = true
    arrivedAt = toP; isTeleporting = false
    if toP.isReturnPortal then
        returnPosition = nil; currentDestination = "Normal"
        notify("Возврат", "Вы на Земле!")
    end
end
local function processPortalTunnel(po, other, root)
    if not po or po.isDestroying or not po.part or not po.part.Parent then return end
    local otherAlive = other and not other.isDestroying and other.part and other.part.Parent
    local cf = po.part.CFrame
    local rel = cf:PointToObjectSpace(root.Position)
    local lat = math.sqrt(rel.X*rel.X + rel.Y*rel.Y)
    local inZone = otherAlive and lat < portalRadius() and rel.Z > -7 and rel.Z < 3
    if inZone then activateHole(po)
    else
        if po.holeActive then releaseHole(po, nil) end
        po.justArrived = false
    end
    local prevZ = po.prevZ
    po.prevZ = rel.Z
    if not inZone or isTeleporting then return end
    if os.clock() - lastTeleport < M.DM.COOLDOWN then return end
    local crossed = prevZ ~= nil and prevZ < 0 and rel.Z >= 0
    local noHole = next(po.disabled or {}) == nil
    local touching = noHole and not po.justArrived and rel.Z > -1.6
    if crossed or touching then tunnelTeleport(po, other) end
end
local function processPortal(po, other, root)
    if not po or po.isDestroying or not po.part or not po.part.Parent then return end
    if not other or other.isDestroying or not other.part or not other.part.Parent then return end
    local isH = math.abs(po.normal.Y) > 0.7
    if isH then
        local inside = insidePortalH(po, root)
        if not isTeleporting and inside and not po.wasInsideH then
            if os.clock() - lastTeleport >= M.DM.COOLDOWN then horizontalTeleport(po, other) end
        end
        po.wasInsideH = inside
    else
        processPortalTunnel(po, other, root)
    end
end

shoot = function(gun)
    if not gun or not gun.def or not gun.muzzle or not gun.handle then return end
    if gun.def.noShoot then return end
    gunShotT = os.clock()
    local gunDef = gun.def
    local cM = gun.muzzle
    local cH = gun.handle
    local cam = workspace.CurrentCamera
    if not cam then return end
    local mp = UserInputService:GetMouseLocation()
    local ray = cam:ScreenPointToRay(mp.X, mp.Y)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignoreList()
    local result = workspace:Raycast(ray.Origin, ray.Direction*M.DM.RANGE, params)
    local snd = cH:FindFirstChild("ShotSound")
    if snd then snd:Play() end
    local bc = gunDef.beamColor
    local tx = gunDef.portalTexture
    local pm = gunDef.portalModelId
    if not result then
        createBeam(cM.Position, ray.Origin + ray.Direction*M.DM.RANGE, bc); return
    end
    createBeam(cM.Position, result.Position, bc)
    arrivedAt = nil
    local hp = result.Instance
    local function resetBoth()
        if portal1 then destroyPortalPart(portal1) end
        if portal2 then destroyPortalPart(portal2) end
        portal1, portal2 = nil, nil
    end
    if currentDestination == "Garage" then
        resetBoth()
        portal1 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
        ensureGarageExists()
        local ep = garageSpawnPoint or (M.DM.GARAGE_C + Vector3.new(0,20,0))
        portal2 = createPortalPart(ep, Vector3.new(0,0,-1), false, tx, nil, pm)
        notify("В Гараж!", "Войди!")
    elseif currentDestination == "Retro" then
        resetBoth()
        portal1 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
        ensureRetroExists()
        local gp = findGround(M.DM.RETRO_C, 400)
        portal2 = createExitPortalInDim(gp, result.Normal, tx, pm)
        notify("В RETRO!", "Войди!")
    elseif currentDestination == "Omega" then
        resetBoth()
        portal1 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
        ensureOmegaExists()
        local ep = omegaSpawnPoint or (M.DM.OMEGA_C + Vector3.new(0,20,0))
        portal2 = createPortalPart(ep, Vector3.new(0,0,-1), false, tx, nil, pm)
        notify("В Omega!", "Войди!")
    elseif currentDestination == "Return" then
        if not returnPosition then
            notify("Инфо", "Точка потеряна!"); currentDestination = "Normal"; return
        end
        resetBoth()
        portal1 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
        local ep = returnPosition + Vector3.new(0,3,0)
        portal2 = createPortalPart(ep, Vector3.new(0,0,-1), true, tx, nil, pm)
        notify("На Землю!", "Войди!")
    else
        if nextPortal == 1 then
            if portal1 then destroyPortalPart(portal1) end
            portal1 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
            nextPortal = 2
        else
            if portal2 then destroyPortalPart(portal2) end
            portal2 = createPortalPart(result.Position, result.Normal, false, tx, hp, pm)
            nextPortal = 1
        end
    end
end

    local function simStep(dt)
        dt = math.min(dt, 0.05)
        local list = sim.list
        for i = #list, 1, -1 do
            local q = list[i]
            local part = q.part
            if not part or not part.Parent then
                table.remove(list, i)
            else
                q.age = q.age + dt
                q.vel = q.vel + GRAV*dt
                local from = part.Position
                local step = q.vel*dt
                local hit = nil
                if step.Magnitude > 1e-4 then hit = workspace:Raycast(from, step, q.rp) end
                if hit then
                    table.remove(list, i)
                    q.onHit(q, hit)
                else
                    local np = from + step
                    if q.kind == "drop" then
                        part.CFrame = CFrame.new(np)
                    else
                        q.rot = q.rot * CFrame.Angles(q.spin.X*dt, q.spin.Y*dt, q.spin.Z*dt)
                        part.CFrame = CFrame.new(np) * q.rot
                    end
                    if q.age > q.life or np.Y < -500 then
                        table.remove(list, i); part:Destroy()
                    end
                end
            end
        end
        if #list == 0 and sim.conn then sim.conn:Disconnect(); sim.conn = nil end
    end

        if ActiveScene == scene then ActiveScene = nil end
    end

    local function bez(a, b, c, t)
        local u = 1 - t
        return a*(u*u) + b*(2*u*t) + c*(t*t)
    end

    local function spawnBloodFlow(scene, targetPart, delay, dur)
        task.spawn(function()
            task.wait(delay)
            if not scene or not scene.puddle or not targetPart or not targetPart.Parent then return end
            local P = scene.puddle
            local ang = math.random()*6.283
            local rad = math.random()*P.radius*0.8
            local p0 = P.center + Vector3.new(math.cos(ang)*rad, 0.12, math.sin(ang)*rad)
            local fx = getFxFolder()
            local segN = 8
            local segs = {}
            local w = 0.16 + math.random()*0.12
            for i=1,segN do
                local s = newFxPart(fx, (math.random()<0.5) and M.CLR.BLOOD or M.CLR.BLOOD_LT,
                    Enum.Material.SmoothPlastic, Vector3.new(w,w,0.1), 1)
                s.Reflectance = 0.12
                segs[i] = s
            end
            local wob = randUnit()
            local t0 = os.clock()
            local arrived = false
            while true do
                local el = os.clock() - t0
                if el > dur or not targetPart.Parent then break end
                local p2 = targetPart.Position
                local mid = (p0 + p2)/2 + Vector3.new(0, 2.2 + (p2-p0).Magnitude*0.15, 0) + wob*1.2
                local head = easeOut(el/(dur*0.6))
                local tail = easeIn(math.max(0, el/dur - 0.35)/0.65)
                if tail > head then tail = head end
                local prev = bez(p0, mid, p2, tail)
                for i=1,segN do
                    local u = tail + (head - tail)*(i/segN)
                    local pt = bez(p0, mid, p2, u) + wob*math.sin(u*9 + el*12)*0.22
                    local d = (pt - prev).Magnitude
                    local seg = segs[i]
                    if d < 0.02 then seg.Transparency = 1
                    else
                        seg.Transparency = 0
                        local ww = w*(0.4 + 0.6*i/segN)
                        seg.Size = Vector3.new(ww, ww, d)
                        seg.CFrame = lookCF(prev, pt) * CFrame.new(0,0,-d/2)
                    end
                    prev = pt
                end
                if head >= 1 and not arrived then
                    arrived = true
                    spawnBloodExplosion(targetPart.CFrame, 3, 1.5)
                end
                RunService.RenderStepped:Wait()
            end
            for _, s in ipairs(segs) do if s then s:Destroy() end end
        end)
    end

    local function playRegenStorm(limbName, parts, S)
        local anchor = parts[1]
        local char = LocalPlayer.Character
        if not anchor or not anchor.Parent or not char then return end
        local folder = getFxFolder()
        local isLimb = (limbName ~= "Torso" and limbName ~= "Head")
        local reach = 0
        for _, part in ipairs(parts) do reach = reach + part.Size.Y end
        local originLocal = CFrame.new()
        if isLimb then originLocal = CFrame.new(0, anchor.Size.Y/2 - 0.25, 0)
        else reach = math.max(anchor.Size.X, anchor.Size.Y, anchor.Size.Z) * ((limbName=="Head") and 1.2 or 0.9) end
        local function originCF() return anchor.CFrame * originLocal end
        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Exclude
        rp.FilterDescendantsInstances = {char, folder}
        rp.IgnoreWater = true
        local o0 = originCF()
        local charRoot = char:FindFirstChild("HumanoidRootPart")
        local center = charRoot and charRoot.Position or o0.Position
        local outward = Vector3.new(o0.Position.X-center.X, 0, o0.Position.Z-center.Z)
        outward = (outward.Magnitude > 0.1) and outward.Unit or nil
        local function pickGround()
            local off
            if outward and isLimb then
                local side = Vector3.new(-outward.Z, 0, outward.X)
                off = outward*(0.4 + math.random()*3.2) + side*((math.random()-0.5)*2.4)
            else
                local ang = math.random()*math.pi*2
                local dist = 0.5 + math.random()*3.0
                off = Vector3.new(math.cos(ang)*dist, 0, math.sin(ang)*dist)
            end
            local from = o0.Position + off
            local hit = workspace:Raycast(from, Vector3.new(0,-30,0), rp)
            if hit then return hit.Position, hit.Normal end
            return from - Vector3.new(0,3,0), Vector3.new(0,1,0)
        end
        local function spawnSplatStorm(pos, norm)
            for i=1, math.random(1,2) do
                local sz = 0.45 + math.random()*1.0
                local part = newFxPart(folder, M.CLR.RED,
                    Enum.Material.SmoothPlastic,
                    Vector3.new(sz, 0.06, sz*(0.6+math.random()*0.8)), 0)
                local base = lookCF(pos, pos+norm) * CFrame.Angles(-math.pi/2, 0, 0)
                part.CFrame = base
                    * CFrame.new((math.random()-0.5)*0.9, 0.03 + i*0.005, (math.random()-0.5)*0.9)
                    * CFrame.Angles(0, math.random()*math.pi*2, 0)
                task.delay(M.TM.SPLAT_LIFE/S, function()
                    if part and part.Parent then
                        TweenService:Create(part, TweenInfo.new(M.TM.SPLAT_FADE, Enum.EasingStyle.Quad), {Transparency=1}):Play()
                        task.delay(M.TM.SPLAT_FADE+0.05, function() if part then part:Destroy() end end)
                    end
                end)
            end
        end
        local strands = {}

        local function tryBuildModelSeg(modelKey)
            if not modelKey then return nil end
            local m = buildRegenModel(modelKey)
            if not m then return nil end
            m.Parent = folder
            hideRegenModel(m)
            return m
        end

        local function addStrand(kind)
            local s = {
                dir = randUnit(),
                kinks = {randUnit(), randUnit(), randUnit()},
                phase = math.random()*6.283,
                freq = 7 + math.random()*8,
                delay = math.random()*0.05,
                sDelay = math.random()*M.TM.STREAK_JIT,
                segs = {}, segN=3, baseTr=0, hasStreak=false,
                modelKey = nil,
            }
            local color, material = M.CLR.NERVE_A, Enum.Material.SmoothPlastic
            if kind == "blue" then
                color = (math.random()<0.5) and M.CLR.NERVE_A or M.CLR.NERVE_B
                s.w, s.h = 0.22 + math.random()*0.1, 0.14 + math.random()*0.05
                s.len = reach*(0.6 + math.random()*0.5)
                s.hasStreak = false
                s.modelKey = "muscleBlue"
            elseif kind == "brown" then
                color = (math.random()<0.5) and M.CLR.MUSCLE_A or M.CLR.MUSCLE_B
                s.w, s.h = 0.22 + math.random()*0.1, 0.14 + math.random()*0.05
                s.len = reach*(0.6 + math.random()*0.5)
                s.hasStreak = false
                s.modelKey = "muscleRed"
            elseif kind == "bone" then
                color = M.CLR.BONE
                s.w, s.h = 0.3, 0.05
                s.segN = 1; s.len = reach*(0.4 + math.random()*0.4)
                s.modelKey = "bone"
            end
            s.color = color
            for k=1,s.segN do
                local m = tryBuildModelSeg(s.modelKey)
                if m then
                    s.segs[k] = {model = m, cfgKey = s.modelKey}
                else
                    s.segs[k] = {part = newFxPart(folder, color, material, Vector3.new(s.w, s.h, 0.1), 1)}
                end
            end
            if s.hasStreak then s.target, s.normal = pickGround() end
            table.insert(strands, s)
        end
        local nB, nBr, nBn
        if isLimb then nB, nBr, nBn = 3, 3, 2
        elseif limbName == "Head" then nB, nBr, nBn = 3, 3, 2
        else nB, nBr, nBn = 5, 5, 2 end
        for _=1,nB do addStrand("blue") end
        for _=1,nBr do addStrand("brown") end
        for _=1,nBn do addStrand("bone") end

        local rootModel = buildRegenModel("root")
        if rootModel then
            rootModel.Parent = folder
            local base = originCF()
            rootModel:PivotTo(regenModelCF("root", base))
            task.delay((M.TM.TOTAL + M.TM.STREAK_JIT + 0.2)/S, function()
                if rootModel then rootModel:Destroy() end
            end)
        end

        spawnBloodExplosion(o0, 10, 3.5)
        task.delay((M.TM.CHAOS+0.05)/S, function() spawnBloodExplosion(originCF(), 5, 2.5) end)
        local turnStart = M.TM.CHAOS
        local streakStart = M.TM.CHAOS + M.TM.TURN

        local function killStrand(s)
            s.dead = true
            for _, seg in ipairs(s.segs) do
                if seg.model then pcall(function() seg.model:Destroy() end) end
                if seg.part then pcall(function() seg.part:Destroy() end) end
            end
        end

        local function updateStrand(s, oCF, t)
            if s.dead then return end
            local origin = oCF.Position
            if s.hasStreak and t >= streakStart + s.sDelay then
                local v = (t - streakStart - s.sDelay)/M.TM.STREAK
                if v >= 1 then killStrand(s); return end
                local seg = s.segs[1]
                if not s.streakInit then
                    s.streakInit = true
                    for k=2,#s.segs do
                        local sm = s.segs[k]
                        if sm.part then sm.part.Transparency = 1 end
                        if sm.model then hideRegenModel(sm.model) end
                    end
                    if seg.part then seg.part.Color = s.color; seg.part.Transparency = 0 end
                end
                local tip = origin:Lerp(s.target, easeOut(v))
                local tail = origin:Lerp(s.target, easeIn(math.max(v-0.18,0)/0.82))
                local d = (tip-tail).Magnitude
                if seg.part then
                    if d < 0.03 then seg.part.Transparency = 1
                    else
                        seg.part.Transparency = 0
                        seg.part.Size = Vector3.new(s.w*1.1*(1-0.55*v), s.h*0.6, d)
                        seg.part.CFrame = lookCF(tail, tip)*CFrame.new(0,0,-d/2)
                    end
                elseif seg.model then
                    if d < 0.03 then hideRegenModel(seg.model)
                    else
                        local baseCF = lookCF(tail, tip)*CFrame.new(0,0,-d/2)
                        seg.model:PivotTo(regenModelCF(seg.cfgKey, baseCF))
                    end
                end
                if v >= 0.8 and not s.splat then s.splat = true; spawnSplatStorm(s.target, s.normal) end
                return
            end
            if not s.hasStreak and t >= streakStart then killStrand(s); return end
            local tt = t - s.delay
            if tt < 0 then return end
            local g = easeOut(tt/0.13)
            local u = math.clamp((t-turnStart)/M.TM.TURN, 0, 1)
            local segLen = (s.len/s.segN)*g*(1-0.3*u)
            local cur = origin
            for k=1,s.segN do
                local wob = math.sin(t*s.freq + s.phase + k*1.7)
                local lv = safeUnit(s.dir + s.kinks[k]*(0.9+0.5*wob), s.dir)
                local wd = oCF:VectorToWorldSpace(lv)
                if u > 0 then wd = safeUnit(wd + Vector3.new(0,-1.4*u,0), wd) end
                local nxt = cur + wd*segLen
                local d = (nxt-cur).Magnitude
                local seg = s.segs[k]
                local baseCF = lookCF(cur, nxt)*CFrame.new(0,0,-d/2)
                if seg.part then
                    if d < 0.02 then seg.part.Transparency = 1
                    else
                        if s.hasStreak then seg.part.Transparency = 0
                        else seg.part.Transparency = s.baseTr + (1-s.baseTr)*u end
                        seg.part.Size = Vector3.new(s.w, s.h, d)
                        seg.part.CFrame = baseCF
                    end
                elseif seg.model then
                    if d < 0.02 then hideRegenModel(seg.model)
                    else
                        seg.model:PivotTo(regenModelCF(seg.cfgKey, baseCF))
                    end
                end
                cur = nxt
            end
        end

        local t0 = os.clock()
        local endT = M.TM.TOTAL + M.TM.STREAK_JIT
        local okLoop, errLoop = pcall(function()
            while true do
                local t = (os.clock()-t0)*S
                if t >= endT then break end
                if not (anchor and anchor.Parent) then break end
                local oCF = originCF()
                for _, s in ipairs(strands) do updateStrand(s, oCF, t) end
                RunService.RenderStepped:Wait()
            end
        end)
        if not okLoop then log("regen storm error:", errLoop) end
        for _, s in ipairs(strands) do if not s.dead then killStrand(s) end end
    end

    local function origTransparency(part)
        local o = originalDataRef[part]
        return (o and o.Transparency) or 0
    end

    local function runPrintSequence(limbName, parts)
        task.spawn(function()
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not char or not hum then busyLimbsRef[limbName] = false; return end
            local anchorPart = parts[1]
            local S = math.max(config.RegenSpeed or 1, 0.05)
            local scene = ActiveScene
            local rp = makeRP(char)
            local refl0 = {}
            startPrintSound(anchorPart, M.TM.TOTAL/S)
            cameraShake(0.08, 0.25)
            for _, part in ipairs(parts) do
                if originalDataRef[part] == nil then
                    originalDataRef[part] = {
                        Color=part.Color, Material=part.Material,
                        Transparency=part.Transparency, CanCollide=part.CanCollide,
                    }
                end
                refl0[part] = part.Reflectance
                part.Transparency = 1
            end

            if scene and scene.puddle then
                local nFlows = (limbName == "Torso") and 3 or 2
                for i=1,nFlows do
                    local tgt = parts[((i-1) % #parts) + 1]
                    spawnBloodFlow(scene, tgt, (i-1)*0.07/S, (0.55 + math.random()*0.2)/S)
                end
            end

            task.delay(M.TM.GHOST_START/S, function()
                task.spawn(function()
                    task.spawn(function()
                        for _=1, #parts do
                            local part = parts[math.random(1,#parts)]
                            if part and part.Parent then
                                local o = part.Position + Vector3.new((math.random()-0.5)*part.Size.X, -part.Size.Y*0.3, (math.random()-0.5)*part.Size.Z)
                                spawnDroplet(scene, rp, o, Vector3.new((math.random()-0.5)*2, -2, (math.random()-0.5)*2),
                                    0.1 + math.random()*0.1)
                            end
                            task.wait(0.06/S)
                        end
                    end)
                    for _, tr in ipairs({0.9, 0.55, 0.88, 0.7}) do
                        for _, part in ipairs(parts) do
                            if part and part.Parent then part.Transparency = tr end
                        end
                        task.wait(0.035/S)
                    end
                    local dur = (M.TM.GHOST_TIME-0.14)/S + 0.12
                    for _, part in ipairs(parts) do
                        if part and part.Parent then
                            local o = originalDataRef[part]
                            local goal = {Transparency = origTransparency(part), Reflectance = refl0[part] or 0}
                            if o and o.Color then goal.Color = o.Color end
                            TweenService:Create(part, TweenInfo.new(dur, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), goal):Play()
                        end
                    end
                end)
            end)

            local ok, err = pcall(playRegenStorm, limbName, parts, S)
            if not ok then log("playRegenStorm ошибка:", err) end
            task.wait(0.05/S)
            endPrintSound()
            for _, part in ipairs(parts) do
                if part and part.Parent then
                    part.Transparency = origTransparency(part)
                    local o = originalDataRef[part]
                    if o then
                        if o.CanCollide ~= nil then part.CanCollide = o.CanCollide end
                        if o.Color then part.Color = o.Color end
                        if o.Material then part.Material = o.Material end
                    end
                    part.Reflectance = refl0[part] or 0
                end
            end
            busyLimbsRef[limbName] = false
        end)
    end

        if not hum then return end
        local t = (hum.RigType == Enum.HumanoidRigType.R6) and M.LB.R6 or M.LB.R15
        local partsInOrder = t[limbName]
        if not partsInOrder then return end
        local parts = {}
        for _, n in ipairs(partsInOrder) do
            local part = char:FindFirstChild(n)
            if part then
                if originalDataRef[part] == nil then
                    originalDataRef[part] = {
                        Color=part.Color, Material=part.Material,
                        Transparency=part.Transparency, CanCollide=part.CanCollide,

                pcall(function()

                        (math.random()-0.5)*25,
                        (math.random()-0.5)*25)
                    clone.Parent = workspace
                    task.delay(4, function()
                        if clone and clone.Parent then
                            local tw = TweenService:Create(clone, TweenInfo.new(0.6, Enum.EasingStyle.Quad), {Transparency=1})
                            tw:Play()

                        end

            runPrintSequence(limbName, parts)

    getgenv().tryLoseLimbs = function(count)
        count = math.max(1, count or 1)
        if ShatterActive then return false end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return false end
        local t = (hum.RigType == Enum.HumanoidRigType.R6) and M.LB.R6 or M.LB.R15
        local candidates = {}
        for _, name in ipairs({"RightArm","LeftArm","RightLeg","LeftLeg"}) do
            if not busyLimbsRef[name] then
                local pIn = t[name]
                if pIn then
                    local visible = false
                    for _, n in ipairs(pIn) do
                        local part = char:FindFirstChild(n)
                        if part and part.Transparency < 0.9 then visible = true; break end
                    end
                    if visible then table.insert(candidates, name) end
                end
            end
        end
        if #candidates == 0 then return false end
        local lost = 0
        for i = 1, math.min(count, #candidates) do
            local idx = math.random(1, #candidates)
            getgenv().loseLimb(candidates[idx])
            table.remove(candidates, idx)
            lost = lost + 1
        end
        return lost > 0
    end
    getgenv().tryLoseRandomLimb = function()
        return getgenv().tryLoseLimbs(1)
    end

    local function celebrateRegenComplete()
        local char = LocalPlayer.Character
        local head = char and char:FindFirstChild("Head")
        if not head then return end

        local model = loadAssetModel(M.SND.DECAL)
        if not model then return end
        if not model:IsA("Model") then
            local w = Instance.new("Model")
            model.Parent = w
            model = w
        end
        model.Parent = workspace

        for _, d in ipairs(model:GetDescendants()) do
            if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript") then
                d:Destroy()
            elseif d:IsA("BasePart") then
                d.Anchored = true
                d.CanCollide = false
                d.CanTouch = false
                d.CanQuery = false
                d.CastShadow = false
            end
        end

        local cfg = (getgenv().RegenModelConfig or {}).decal or {}
        pcall(function() model:ScaleTo(cfg.scale or 1) end)

        local conn = RunService.RenderStepped:Connect(function()
            if not head or not head.Parent or not model or not model.Parent then return end
            local cam = workspace.CurrentCamera
            if not cam then return end
            local camPos = cam.CFrame.Position
            local basePos = head.Position + Vector3.new(0, 2.2, 0)
            local cf = CFrame.lookAt(basePos, camPos)
            local off = CFrame.new(cfg.offX or 0, cfg.offY or 0, cfg.offZ or 0)
                * CFrame.Angles(math.rad(cfg.rotX or 0), math.rad(cfg.rotY or 0), math.rad(cfg.rotZ or 0))
            pcall(function() model:PivotTo(cf * off) end)
        end)

        task.delay(3, function()
            if conn then conn:Disconnect() end
            if model then model:Destroy() end
        end)
    end

    local hiddenExtras = setmetatable({}, {__mode="k"})

    local function hideObj(obj)
        if isInsideTool(obj) then return end
        if obj:IsA("BodyColors") then return end

        if obj:IsA("BasePart") and isAccessoryHandle(obj) then
            if hiddenExtras[obj] == nil then hiddenExtras[obj] = obj.Transparency end
            obj.Transparency = 1
            return
        end
        if obj:IsA("Decal") and isInsideAccessory(obj) then
            if hiddenExtras[obj] == nil then hiddenExtras[obj] = obj.Transparency end
            obj.Transparency = 1
            return
        end
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
            if originalDataRef[obj]==nil then
                originalDataRef[obj] = {
                    Color=obj.Color, Material=obj.Material,
                    Transparency=obj.Transparency, CanCollide=obj.CanCollide,
                }
            end
            obj.Transparency = 1
        elseif obj:IsA("Decal") and not isClothingDecal(obj) then
            if hiddenExtras[obj]==nil then hiddenExtras[obj]=obj.Transparency end
            obj.Transparency = 1
        end
    end
    local function hideWholeBody(char)
        if not char then return end
        for _, obj in ipairs(char:GetDescendants()) do hideObj(obj) end
    end
    local function restoreExtras()
        for inst, tr in pairs(hiddenExtras) do
            if inst and inst.Parent then pcall(function() inst.Transparency = tr end) end
            hiddenExtras[inst] = nil
        end
    end

        end
    end

    local function spawnBloodTransformation(char, originPos, originCF)
        local scene = newScene()
        local rp = makeRP(char)
        scene.rp = rp
        ActiveScene = scene

        local hit = workspace:Raycast(originPos, Vector3.new(0, -400, 0), rp)
        local groundPos = hit and hit.Position or (originPos - Vector3.new(0, 3, 0))
        local groundN = hit and hit.Normal or Vector3.new(0,1,0)
        if groundN.Y < 0.5 then groundN = Vector3.new(0,1,0) end
        scene.ground = groundPos

        lightFlash(originPos, M.CLR.BLOOD_LT, 8, 26, 0.6)

        for _, obj in ipairs(char:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" and not isInsideTool(obj) then
                spawnBloodExplosion(obj.CFrame, 3, 4)
            end
        end
        spawnBloodExplosion(originCF, 25, 6)
        task.delay(0.08, function() spawnBloodExplosion(originCF * CFrame.new(0,1,0), 15, 5) end)
        task.delay(0.16, function() spawnBloodExplosion(originCF, 12, 4) end)

        wallSplatter(scene, rp, originPos + Vector3.new(0,1,0), 8)

        task.spawn(function()
            for _=1,8 do
                for _=1,2 do
                    spawnDroplet(scene, rp,
                        originPos + Vector3.new((math.random()-0.5)*2, math.random()*3 + 0.5, (math.random()-0.5)*2),
                        Vector3.new((math.random()-0.5)*6, -5, (math.random()-0.5)*6),
                        0.26 + math.random()*0.26)
                end
                task.wait(0.05)
            end
        end)

        task.delay(0.3, function() createPuddle(scene, groundPos, groundN) end)
        return scene
    end

    local function scheduleFullRegen()
        local S = math.max(config.RegenSpeed or 1, 0.05)
        local stepDelay = 0.22/S
        local oneRegenTime = (M.TM.TOTAL + M.TM.STREAK_JIT)/S
        for i, limbName in ipairs(M.LB.REGEN_ORDER) do
            task.delay(stepDelay*(i-1), function()
                busyLimbsRef[limbName] = false
                if limbName == "Head" then
                    if getgenv().regenerateHead then getgenv().regenerateHead() end
                elseif limbName == "Torso" then
                    if getgenv().regenerateTorso then getgenv().regenerateTorso() end
                else
                    if getgenv().regenerateLimb then getgenv().regenerateLimb(limbName) end
                end
            end)
        end
        local headIdx = table.find(M.LB.REGEN_ORDER, "Head") or #M.LB.REGEN_ORDER
        task.delay(stepDelay*(headIdx-1) + oneRegenTime*0.6, restoreExtras)
        local total = stepDelay*(#M.LB.REGEN_ORDER - 1) + oneRegenTime
        if ActiveScene then sceneDrain(ActiveScene, total*0.9) end
        return total
    end

    local function emergeFromGround(root, rootCF, dur, scene)
        if not root or not root.Parent then return end
        local OFF, TILT = 3.4, 38
        local baseRot = rootCF - rootCF.Position
        local pos0 = rootCF.Position
        local gp = (scene and scene.ground) or (pos0 - Vector3.new(0,3,0))
        local t0 = os.clock()
        local nextFx = 0
        while root.Parent do
            local el = os.clock() - t0
            local u = el/dur
            if u >= 1 then break end
            local e = easeOut(u)
            local shake = (1-u)*0.14
            root.CFrame = CFrame.new(pos0 + Vector3.new((math.random()-0.5)*shake, -OFF*(1-e), (math.random()-0.5)*shake))
                * baseRot * CFrame.Angles(-math.rad(TILT*(1-e)^1.5), 0, 0)
            if el >= nextFx then
                nextFx = el + 0.09
                spawnBloodExplosion(CFrame.new(gp + Vector3.new(0,0.4,0)), 3, 1.6)
            end
            RunService.Heartbeat:Wait()
        end
        if root.Parent then root.CFrame = rootCF end
    end

    getgenv().shatterBody = function()
        if ShatterActive then return end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not char or not hum or not root or hum.Health <= 0 then return end
        if flight.active then stopFlight(true) end
        ShatterActive = true; getgenv().ShatterActive = true

        local oldWalk = hum.WalkSpeed
        local oldJumpP = hum.JumpPower
        local oldJumpH
        pcall(function() oldJumpH = hum.JumpHeight end)
        local oldAnchored = root.Anchored
        hum.WalkSpeed = 0
        pcall(function() hum.JumpPower = 0 end)
        pcall(function() hum.JumpHeight = 0 end)
        root.Anchored = true
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero

        playDisintegrateSound(root)
        cameraShake(0.4, 0.7)
        screenFlash(Color3.fromRGB(200,20,20), 0.2, 0.4)

        task.spawn(function()
            local rootCF = root.CFrame
            local rootPos = root.Position

            local scene = spawnBloodTransformation(char, rootPos, rootCF)
            hideWholeBody(char)

            task.delay(1.6, function() sceneCoagulate(scene, 2.2) end)

            task.wait(M.TM.FRAG_LIFE)

            pcall(regenBurst, char)
            local totalTime = scheduleFullRegen()
            task.spawn(emergeFromGround, root, rootCF, totalTime*0.85, scene)
            task.wait(totalTime)
            restoreExtras()

            pcall(regenCompleteVFX, char)
            sceneClear(scene, 3)

            hum.WalkSpeed = oldWalk
            pcall(function() hum.JumpPower = oldJumpP end)
            pcall(function() if oldJumpH then hum.JumpHeight = oldJumpH end end)
            if root and root.Parent then root.Anchored = oldAnchored end

            pcall(celebrateRegenComplete)

            ShatterActive = false; getgenv().ShatterActive = false
        end)
    end

    local function handleNaturalDeathShatter(character)
        if ShatterActive then return end
        local hum = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        if hum and hum.Health > 0 then return end
        ShatterActive = true; getgenv().ShatterActive = true
        PendingHideOnSpawn = true; getgenv().PendingHideOnSpawn = true
        local deathCF = root.CFrame
        local deathPos = root.Position

        local cam = workspace.CurrentCamera
        lockDeathCamera(cam and cam.CFrame or CFrame.lookAt(deathPos + Vector3.new(0,6,12), deathPos))

        cameraShake(0.4, 0.7)
        screenFlash(Color3.fromRGB(200,20,20), 0.2, 0.4)
        playDisintegrateSound(root)
        local scene = spawnBloodTransformation(character, deathPos, deathCF)
        hideWholeBody(character)
        for _, obj in ipairs(character:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" and not isInsideTool(obj) then
                obj.CanCollide = false
            end
        end

        local newChar, gotRespawn = nil, false
        local pinConns, hideConn = {}, nil
        local function stopPin()
            for _, c in ipairs(pinConns) do pcall(function() c:Disconnect() end) end
            pinConns = {}
        end
        local function pinTo(c)
            task.spawn(function()
                local r = c:WaitForChild("HumanoidRootPart", 10)
                if not r then return end
                local function pin()
                    if r.Parent then
                        r.CFrame = deathCF
                        r.AssemblyLinearVelocity = Vector3.zero
                        r.AssemblyAngularVelocity = Vector3.zero
                    end
                end
                r.Anchored = true
                pin()
                table.insert(pinConns, RunService.Stepped:Connect(pin))
                table.insert(pinConns, RunService.Heartbeat:Connect(pin))
                table.insert(pinConns, RunService.RenderStepped:Connect(pin))
            end)
        end

        local rc = LocalPlayer.CharacterAdded:Connect(function(c)
            newChar = c; gotRespawn = true
            pinTo(c)
            hideWholeBody(c)
            hideConn = c.DescendantAdded:Connect(hideObj)
        end)
        local t0 = os.clock()
        while not gotRespawn and os.clock()-t0 < 20 do task.wait(0.05) end
        rc:Disconnect()

        if newChar then
            local nr = newChar:WaitForChild("HumanoidRootPart", 10)
            local nHum = newChar:WaitForChild("Humanoid", 10)
            local oldWalk, oldJumpP, oldJumpH
            if nHum then
                oldWalk = nHum.WalkSpeed
                oldJumpP = nHum.JumpPower
                pcall(function() oldJumpH = nHum.JumpHeight end)
                nHum.WalkSpeed = 0
                pcall(function() nHum.JumpPower = 0 end)
                pcall(function() nHum.JumpHeight = 0 end)
            end
            if nr then nr.Anchored = true end
            hideWholeBody(newChar)
            sceneCoagulate(scene, 1.5)

            task.wait(0.3)
            if nr then nr.CFrame = deathCF end
            stopPin()

            pcall(regenBurst, newChar)
            local totalTime = scheduleFullRegen()
            if nr then task.spawn(emergeFromGround, nr, deathCF, totalTime*0.85, scene) end
            task.wait(totalTime)
            restoreExtras()
            if hideConn then hideConn:Disconnect(); hideConn = nil end

            pcall(regenCompleteVFX, newChar)
            sceneClear(scene, 3)
            if nHum then
                nHum.WalkSpeed = oldWalk
                pcall(function() nHum.JumpPower = oldJumpP end)
                pcall(function() if oldJumpH then nHum.JumpHeight = oldJumpH end end)
            end
            if nr and nr.Parent then nr.Anchored = false end
            pcall(applyWalkMode)
            unlockDeathCamera(nHum)
            pcall(celebrateRegenComplete)
        else
            stopPin()
            if hideConn then hideConn:Disconnect(); hideConn = nil end
            sceneClear(scene, 3)
            unlockDeathCamera(nil)
        end
        PendingHideOnSpawn = false; getgenv().PendingHideOnSpawn = false
        ShatterActive = false; getgenv().ShatterActive = false
    end

    local function setupCharacter(character)
        local hum = character:WaitForChild("Humanoid", 10)
        if not hum then return end
        hideForceFields(character)
        for part in pairs(originalDataRef) do
            if not part.Parent then originalDataRef[part] = nil end
        end
        local t = (hum.RigType == Enum.HumanoidRigType.R6) and M.LB.R6 or M.LB.R15
        for _, parts in pairs(t) do
            for _, n in ipairs(parts) do
                local part = character:WaitForChild(n, 5)
                if part and originalDataRef[part] == nil then
                    originalDataRef[part] = {
                        Color=part.Color, Material=part.Material,
                        Transparency=part.Transparency, CanCollide=part.CanCollide,
                    }
                end
            end
        end
        if PendingHideOnSpawn then
            PendingHideOnSpawn = false; getgenv().PendingHideOnSpawn = false
            hideWholeBody(character)
        end
        if config.RespawnAtDeathSpot then
            track(hum.Died:Connect(function()
                if ShatterActive then return end
                task.spawn(function() handleNaturalDeathShatter(character) end)
            end))
        end

        if config.LimbLossRegen then
            local lastHealth = hum.Health
            local lastLimbLoss = 0
            local threshold = config.LimbLossHP or 15
            local cooldown = config.LimbLossCooldown or 1.5
            track(hum.HealthChanged:Connect(function(hp)
                if hp <= 0 then lastHealth = hp; return end
                if ShatterActive then lastHealth = hp; return end
                if hp < lastHealth then
                    local dmg = lastHealth - hp
                    lastHealth = hp
                    if dmg >= threshold and (os.clock() - lastLimbLoss) >= cooldown then
                        lastLimbLoss = os.clock()
                        local numLimbs = 1
                        if dmg >= threshold * 2.5 then numLimbs = 2 end
                        if getgenv().tryLoseLimbs then
                            getgenv().tryLoseLimbs(numLimbs)
                        end
                    end
                else
                    lastHealth = hp
                end
            end))
        end
    end
    if LocalPlayer.Character then setupCharacter(LocalPlayer.Character) end
    track(LocalPlayer.CharacterAdded:Connect(setupCharacter))
end
